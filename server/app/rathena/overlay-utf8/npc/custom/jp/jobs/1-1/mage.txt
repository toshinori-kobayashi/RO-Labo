				}
			}
			else {
				mes "[調合機]";
				mes "シリアル番号は#" + .@input + "で間違いないか？";
				next;
				if (select("確認する。:キャンセル。") == 1) {
					if (.@input == 8472)
						set .@magic_powder,1;
					else if (.@input == 3735)
						set .@magic_powder,2;
					else if (.@input == 2750)
						set .@magic_powder,3;
					else if (.@input == 5429)
						set .@magic_powder,4;
					else
						set .@magic_powder,5;
				}
				break;
			}
		}
		mes "[調合機]";
		mes "触媒となる";
		mes "宝石を選んでくれ。";
		next;
		switch(select("黄色いジェムストーン。:赤いジェムストーン。:青いジェムストーン。:1カラットダイヤモンド。:スキップ。")) {
		case 1: set .@mixitem3,1; break;
		case 2: set .@mixitem3,2; break;
		case 3: set .@mixitem3,3; break;
		case 4: set .@mixitem3,4; break;
		case 5: break;
		}
		mes "[調合機]";
		mes "準備完了。";
		mes "調合処理を";
		mes "開始する。";
		mes "少し待ってくれ。";
		next;
		mes "[調合機]";
		mes "- 本日の格言 -";
		switch(rand(1,5)) {
		case 1:
			mes "目には目を：人から奪ったなら、何らかの形で返すか償わなければならない。";
			break;
		case 2:
			mes "信用は人の通貨：本物の信頼には、計り知れない価値がある。";
			break;
		case 3:
			mes "因果応報：結局のところ、人は自分が他人にした通りの扱いを受けるものだ。";
			mes "つまり「他人を傷つければ、避けられない形で自分も傷つけられる」ということだ。";
			break;
		case 4:
			mes "遠くの親戚より近くの他人：助けが必要なとき、頼りになるのは身近な人たちだ。";
			break;
		default:
			mes "類は友を呼ぶ：友人を見れば、その人の人柄がわかるものだ。";
		}
		next;
		if (.@mixitem1_1 == 2 && .@mixitem1_2 == 3 && .@mixitem1_3 == 1 && .@mixitem2 == 1 && .@mixitem3 == 1 && .@magic_powder == 1) {
			mes "[調合機]";
			mes "マジシャン試験溶液 1号。";
			getitem 1071,1; //Mage_Test_1
			next;
		}
		else if (.@mixitem1_1 == 3 && .@mixitem1_2 == 1 && .@mixitem1_3 == 1 && .@mixitem2 == 0 && .@mixitem3 == 2 && .@magic_powder == 2) {
			mes "[調合機]";
			mes "マジシャン試験溶液 2号。";
			getitem 1085,1; //Mage_Test_2
			next;
		}
		else if (.@mixitem1_1 == 6 && .@mixitem1_2 == 1 && .@mixitem1_3 == 0 && .@mixitem2 == 1 && .@mixitem3 == 3 && .@magic_powder == 3) {
			mes "[調合機]";
			mes "マジシャン試験溶液 3号。";
			getitem 1086,1; //Mage_Test_3
			next;
		}
		else if (.@mixitem1_1 == 2 && .@mixitem1_2 == 3 && .@mixitem1_3 == 0 && .@mixitem2 == 2 && .@mixitem3 == 4 && .@magic_powder == 4) {
			mes "[調合機]";
			mes "マジシャン試験溶液 4号。";
			getitem 1087,1; //Mage_Test_4
			next;
		}
		else {
			mes "[調合機]";
			mes "予期しない";
			mes "エラーが発生した。";
			getitem 1090,1; //Mage_Test_Etc
			next;
		}
		mes "[調合機]";
		mes "調合完了。";
		mes "ありがとう。";
		close;
	}
	close;
}
