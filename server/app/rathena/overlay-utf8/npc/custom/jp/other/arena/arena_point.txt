//= Point exchanger: Turbo track points -> Arena Points

prt_are_in,103,11,3	script	アリーナポイント管理人::Arena Point Manager	732,{
	mes "[アリーナポイント管理人]";
	mes "アリーナでの活躍は";
	mes "いかがですか？";
	mes "アルデバランのターボトラックで";
	mes "ターボトラックポイントを貯めたなら、";
	mes "アリーナポイントに交換できますよ。";
	next;
	switch(select("ポイント確認:ポイント交換:^660000交換について^000000")) {
	case 1:
		mes "[アリーナポイント管理人]";
		mes "" + strcharinfo(0) + "さん、";
		mes "現在の所持ポイントは";
		mes "" + arena_point + " アリーナポイントと";
		mes "" + tt_point + " ターボトラックポイントです。";
		close;
	case 2:
		mes "[アリーナポイント管理人]";
		mes "" + strcharinfo(0) + "さん、";
		mes "現在の所持ポイントは";
		mes "" + arena_point + " アリーナポイントと";
		mes "" + tt_point + " ターボトラックポイントです。";
		next;
		mes "[アリーナポイント管理人]";
		mes "交換したいレートを";
		mes "下のメニューから";
		mes "選んでください。";
		mes "なお、10ポイントを超えて";
		mes "まとめて交換する場合は、^4D4DFF10の倍数単位^000000でのみ交換できます。";
		next;
		switch(select("2TP→1AP:4TP→2AP:6TP→3AP:8TP→4AP:10TP以上:キャンセル")) {
		case 1:
			callsub S_ExchangePoints,28999,2,1;
		case 2:
			callsub S_ExchangePoints,28998,4,2;
		case 3:
			callsub S_ExchangePoints,28997,6,2;
		case 4:
			callsub S_ExchangePoints,28996,8,4;
		case 5:
			mes "[アリーナポイント管理人]";
			mes "10ターボトラックポイントを";
			mes "何セット交換するか";
			mes "数字で入力してください。";
			mes "入力できる最大値は";
			mes "20です。";
			mes "キャンセルする場合は「^3355FF0^000000」と入力してください。";
			next;
			input .@input;
			if (.@input <= 0) {
				mes "[アリーナポイント管理人]";
				mes "このサービスの利用を";
				mes "キャンセル";
				mes "しました。";
				close;
			}
			else if (.@input > 20) {
				mes "[アリーナポイント管理人]";
				mes "入力された数値が";
				mes "上限を超えています。";
				mes "20以下の数値を入力してください。";
				close;
			}
			else {
				set .@want_point1,10 * .@input;
				set .@want_point,5 * .@input;
				set .@my_arena_all,arena_point + .@want_point;
				set .@my_turbo_all,tt_point - .@want_point1;
				if (.@my_arena_all > 28999) {
					mes "[アリーナポイント管理人]";
					mes "この交換を行うと";
					mes "アリーナポイントの";
					mes "上限である29,000を";
					mes "超えてしまいます。";
					mes "これ以上は";
					mes "交換できません。";
					next;
					mes "[アリーナポイント管理人]";
					mes "一度アリーナポイントを";
					mes "お使いいただいてから、";
					mes "改めてご利用ください。";
					mes "ご利用ありがとうございました。";
					close;
				}
				if (.@my_turbo_all < 0) {
					mes "[アリーナポイント管理人]";
					mes "申し訳ございませんが";
					mes "この交換を行うのに";
					mes "十分な";
					mes "ターボトラックポイントが";
					mes "ありません。";
					close;
				}
				else {
					mes "[アリーナポイント管理人]";
					mes "10ターボトラックポイントを";
					mes "5アリーナポイントへの";
					mes "交換を " + .@input + " 回行いました。";
					mes "合計 " + .@want_point1 + " ターボトラックポイントが";
					mes "" + .@want_point +" アリーナポイントに交換されました。";
					set tt_point, .@my_turbo_all;
					set arena_point, .@my_arena_all;
					next;
					mes "[アリーナポイント管理人]";
					mes "" + strcharinfo(0) + "さん、";
					mes "現在の所持ポイントは";
					mes "^4682B4" + arena_point + "^000000 アリーナポイントと";
					mes "^00688B" + tt_point + "^000000 ターボトラックポイントです。";
					mes "ご利用ありがとうございました。";
					close;
				}
			}
		case 6:
			mes "[アリーナポイント管理人]";
			mes "このサービスの利用を";
			mes "キャンセル";
			mes "しました。";
			close;
		}
	case 3:
		// Again. no dialog for this option.
		close;
	}
	close;

S_ExchangePoints:
	if (arena_point > getarg(0) ) {
		mes "[アリーナポイント管理人]";
		mes "この交換を行うと";
		mes "アリーナポイントの";
		mes "上限である29,000を";
		mes "超えてしまいます。";
		mes "これ以上は";
		mes "交換できません。";
		next;
		mes "[アリーナポイント管理人]";
		mes "一度アリーナポイントを";
		mes "お使いいただいてから、";
		mes "改めてご利用ください。";
		mes "ご利用ありがとうございました。";
		close;
	}
	.@track_point = getarg(1);
	if (tt_point >= .@track_point) {
		.@arena_point = getarg(2);
		mes "[アリーナポイント管理人]";
		mes "" + strcharinfo(0) + "さん、";
		mes ""+ .@track_point +"ターボトラックポイントを";
		mes ""+ .@arena_point +"アリーナポイントに交換しました。";
		tt_point = tt_point - .@track_point;
		arena_point = arena_point + .@arena_point;
		next;
		mes "[アリーナポイント管理人]";
		mes "" + strcharinfo(0) + "さん、";
		mes "現在の所持ポイントは";
		mes "^4682B4" + arena_point + "^000000 アリーナポイントと";
		mes "^00688B" + tt_point + "^000000 ターボトラックポイントです。";
		mes "ご利用ありがとうございました。";
		close;
	}
	else {
		mes "[アリーナポイント管理人]";
		mes "申し訳ございませんが、";
		mes "ターボトラックポイントが不足しています。";
		mes "この交換には最低でも2ターボトラックポイントが必要です。";
		close;
	}
}
