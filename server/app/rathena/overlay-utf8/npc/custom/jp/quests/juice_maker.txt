payon_in03,188,146,5	script	商人マークス・ハンセン#juice::Marx Hansen#juice	86,{
	if (checkweight(1201,1) == 0) {
		mes "- ちょっと待って！ -";
		mes "- 今のあなたは -";
		mes "- 荷物を持ちすぎています。 -";
		mes "- また今度 -";
		mes "- 倉庫に預けてから来てください。 -";
		close;
	}
	if (MISC_QUEST&1) {
		mes "[商人マークス・ハンセン]";
		mes "いらっしゃい。";
		mes "君も果物を";
		mes "加工しに来たのかい？";
		next;
		switch(select("ジュースを作る。:果物の加工について話を聞く。:やめる。")) {
		case 1:
			mes "[商人マークス・ハンセン]";
			mes "どんなフルーツジュースを作りたいんだい？";
			next;
			switch(select("リンゴジュース:バナナジュース:ニンジンジュース:グレープジュース:キャンセル")) {
			case 1:
				set .@fruit,512;
				set .@juice,531;
				break;
			case 2:
				set .@fruit,513;
				set .@juice,532;
				break;
			case 3:
				set .@fruit,515;
				set .@juice,534;
				break;
			case 4:
				set .@fruit,514;
				set .@juice,533;
				break;
			case 5:
				mes "[商人マークス・ハンセン]";
				mes "それじゃあ…";
				mes "また今度な。";
				close;
			}
			if (countitem(.@fruit) == 0 || countitem(713) == 0 || Zeny < 3) {
				mes "[商人マークス・ハンセン]";
				mes "おっと…";
				mes "材料が足りないみたいだね。"+getitemname(.@juice)+"を作るには、"+getitemname(.@fruit)+"を1個と、空き瓶を1個用意してほしい。手数料として3Zenyも必要だ。";
				next;
				mes "[商人マークス・ハンセン]";
				mes "全部揃ったら、果物をブレンドして美味しいジュースを作ってあげるよ。";
				close;
			}

			mes "[商人マークス・ハンセン]";
			mes "いくつ欲しい？";
			next;
			switch(select("作れるだけ作る。:数を指定する。:やめる。")) {
			case 1:
				set .@make,countitem(.@fruit);
				if (countitem(713) < .@make) set .@make, countitem(713);
				if (Zeny/3 < .@make) set .@make, Zeny/3;
				break;
			case 2:
				mes "[商人マークス・ハンセン]";
				mes "100本より少ない数を選んでくれ。作らないなら「0」と入力してくれ。最大で" + countitem(.@fruit) + "本まで作れるよ。";
				next;
				while(1) {
					input .@input;
					if (.@input == 0) {
						mes "[商人マークス・ハンセン]";
						mes "それじゃあ…";
						mes "また来てくれ。";
						close;
					}
					else if (.@input > 100) {
						mes "[商人マークス・ハンセン]";
						mes "100本を超える数は作れない。別の数を選んでくれ。";
						next;
					}
					else {
						break;
					}
				}
				set .@make,.@input;
				break;
			case 3:
				mes "[商人マークス・ハンセン]";
				mes "それじゃあ…";
				mes "また来てくれ。";
				close;
			}

			set .@total_zeny,3 * .@make;

			if (countitem(.@fruit) < .@make || countitem(713) < .@make || Zeny < .@total_zeny) {
				mes "[商人マークス・ハンセン]";
				mes "おっと…";
				mes "材料が足りないみたいだ。これじゃあどうしようもない。必要な物を集めてきてくれ。";
				close;
			}
			delitem .@fruit,.@make;
			delitem 713,.@make;
			set Zeny, Zeny-.@total_zeny;
			getitem .@juice,.@make;

			mes "[商人マークス・ハンセン]";
			mes "はい、どうぞ！約束通り、新鮮で美味しいジュースだよ。とてもさっぱりしていて飲みやすいはずだ。";
			next;
			mes "[商人マークス・ハンセン]";
			mes "それじゃあ…";
			mes "また来てくれ。";
			close;
		case 2:
			mes "[商人マークス・ハンセン]";
			mes "人間がこれほど大きな社会を築く前は、木から果物を集めて生き延びていたんだ。果物は、我々がこの世に存在できるようにしてくれた、自然からの恵みだったのかもしれないね。";
			next;
			mes "[商人マークス・ハンセン]";
			mes "暮らしが豊かになった今、若い世代は果物を食べなくなったように思う。だから、もっと食べやすくする方法を考え始めたんだ。";
			next;
			mes "[商人マークス・ハンセン]";
			mes "フルーツジュースにすれば、食べやすくなるし、味もずっと良くなると気づいてね。";
			next;
			mes "# フルーツジュース案内 #";
			mes "^CC4E5C- リンゴジュース -^000000";
			mes "リンゴ1個、空き瓶1個、3Zeny。";
			mes "^E3CF57- バナナジュース -^000000";
			mes "バナナ1個、空き瓶1個、3Zeny。";
			mes "^ED9121- ニンジンジュース -^000000";
			mes "ニンジン1個、空き瓶1個、3Zeny。";
			mes "^CC00FF- グレープジュース -^000000";
			mes "ブドウ1個、空き瓶1個、3Zeny。";
			close;
		case 3:
			mes "[商人マークス・ハンセン]";
			mes "おい！";
			mes "人を訪ねたなら";
			mes "ちゃんと話をしなさい！";
			close;
		}
	}
	else {
		mes "[商人マークス・ハンセン]";
		mes "いらっしゃい。";
		mes "何か用かい？";
		next;
		if (select("話す:やめる") == 1) {
			mes "[商人マークス・ハンセン]";
			mes "人間がこれほど大きな社会を築く前は、木から果物を集めて生き延びていたんだ。果物は、我々がこの世に存在できるようにしてくれた、自然からの恵みだったのかもしれないね。";
			next;
			mes "[商人マークス・ハンセン]";
			mes "暮らしが豊かになった今、若い世代は果物を食べなくなったように思う。だから、もっと食べやすくする方法を考え始めたんだ。";
			next;
			mes "[商人マークス・ハンセン]";
			mes "フルーツジュースにすれば、食べやすくなるし、味もずっと良くなると気づいてね。";
			close;
		}
		mes "[商人マークス・ハンセン]";
		mes "おい！";
		mes "人を訪ねたなら";
		mes "ちゃんと話をしなさい！";
		close;
	}
}
