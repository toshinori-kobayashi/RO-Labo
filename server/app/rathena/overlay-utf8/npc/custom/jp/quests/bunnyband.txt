			mes "分かった～";
			mes "必要な品を";

			next;
			setarray .@Items[0],949,100,706,1,722,1,2213,1;
			for(set .@i,0; .@i<8; set .@i,.@i+2)

					mes "[カプラ職員]";

			mes "よかった、";
			mes "バニーバンドを作るのに";
			mes "必要な物が";
			mes "全部揃ってるね。";
			mes "少し待っててね、";
			mes "今作るから…";

			for(set .@i,0; .@i<8; set .@i,.@i+2)
				if (countitem(.@Items[.@i]) < .@Items[.@i+1]) {
					mes "[カプラ職員]";
					mes "あら？ごめんね、";
					mes "実はこれ、";
					mes "今は作れないみたい。";
					mes ""+callfunc("F_InsertPlural",.@Items[.@i+1],getitemname(.@Items[.@i]));
					mes "が足りなくて、";
					mes "バニーバンドを作れないの…";
					close;
				}
			delitem 949,100; //Feather
			delitem 706,1; //Four_Leaf_Clover
			delitem 722,1; //Scarlet_Jewel
			delitem 2213,1; //Cat_Hairband
			mes "[カプラ職員]";
			mes "はい、どうぞ～";
			mes "完璧なバニーバンドよ！";
			mes "気に入ってもらえたら嬉しいな。";

			next;
			mes "[カプラ職員]";
			mes "この特別イベントに";
			mes "参加してくれて、";
			mes "そしていつも";

		case 2:

		case 3:
