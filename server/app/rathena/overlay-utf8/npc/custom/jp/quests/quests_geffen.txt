		mes "ちょっと";
		mes "うぬぼれているようだ。^000000";
		close;
	}

// Number Match Game :: gef_yagu
//============================================================
geffen,122,77,4	script	ニア#yagu::Nia#yagu	700,{
	mes "[ニア]";
	mes "こんにちは、私はニア。";
	mes "数字合わせ遊びの";
	mes "名人マウォンの";
	mes "五番弟子です。さあ、";
	mes "数字合わせゲームで勝負してみませんか？";
	next;
	switch(select("数字合わせゲームとは？:練習で遊ぶ:本番で遊ぶ:記録を見る:やめる")) {
	case 1:
		mes "[ニア]";
		mes "ああ、そうですね。";
		mes "数字合わせゲームの";
		mes "目的は、私が思い浮かべた";
		mes "数字を当てることです。";
		mes "もちろん、私が選べる数字には";
		mes "ちょっとしたルールがあるんですよ。";
		next;
		mes "[ニア]";
		mes "私が考える数字は";
		mes "すべて3桁で、";
		mes "どの桁にも0は";
		mes "使えません。それに、";
		mes "同じ数字を複数の桁で";
		mes "使うこともできないんです。";
		next;
		mes "[ニア]";
		mes "たとえば、103や112、";
		mes "252、701のような";
		mes "数字は選べません。";
		mes "0が含まれていたり、";
		mes "同じ数字を複数回";
		mes "使っていたりするからです。分かりましたか？";
		next;
		mes "[ニア]";
		mes "さて、推測できるのは";
		mes "5回まで。1回答えるごとに、";
		mes "正しい数字がいくつ";
		mes "含まれているか、そのうち";
		mes "何個が正しい桁にあるかを教えます。";
		next;
		mes "[ニア]";
		mes "たとえば、私の考えた";
		mes "数字が168だとします。";
		mes "最初に678と答えたら、6と8の";
		mes "2つの数字は当たっていますが、";
		mes "正しい桁にあるのは8だけです。";
		next;
		mes "[ニア]";
		mes "まずは練習モードで";
		mes "試してみるといいですよ。";
		mes "慣れてきたら、本番の";
		mes "数字合わせゲームに挑戦して";
		mes "みてください。スコアにはスピードも関係するので忘れずに！";
		next;
		mes "[ニア]";
		mes "ああ、もし数字合わせ";
		mes "ゲームで新記録を出したら、";
		mes "あなたの名前が記録";
		mes "されますよ。名声と、";
		mes "もしかしたら富も手に入る";
		mes "このチャンス、お見逃しなく～";
		close;
	case 2:
		mes "[ニア]";
		mes "よし、練習で";
		mes "数字合わせゲームを";
		mes "やってみましょう。私の数字を当ててみてください。";
		mes "やめたい時は、1000より大きい数字を入力してくださいね。";
		set .@practice,1;
		break;
	case 3:
		mes "[ニア]";
		mes "よし、数字合わせゲームを始めましょう！";
		mes "5回のチャンスで、私が考えた数字を当ててください。";
		mes "やめたい時は、1000より大きい数字を入力してくださいね。";
		set .@practice,0;
		break;
	case 4:
		set .@name_record$, $050908_minus1_yagu$;
		if (.@name_record$ == "") //custom
			set .@name_record$,"(未登録)";
		set .@score_record, $050908_minus1_yagu;
		set .@score_min, .@score_record / 60;
		set .@score_sec, .@score_record % 60;
		mes "[ニア]";
		mes "^ff0000" + .@name_record$ + "^000000さんが";
		mes "数字合わせ";
		mes "ゲームの記録保持者です。タイムは";
		mes "^ff0000" + .@score_min + "分" + .@score_sec + "秒^000000でした。";
		close;
	case 5:
		mes "[ニア]";
		mes "お元気で、冒険者さん。";
		mes "また数字合わせゲームが";
		mes "したくなったら、";
		mes "いつでも話しかけて";
		mes "くださいね。それでは～";
		close;
	}
	next;

	// Generate 3 digits of a number.
	while(1) {
		set .@yagu100, rand(1,9);
		set .@yagu10, rand(1,9);
		set .@yagu1, rand(1,9);
		if ((.@yagu100 != .@yagu10) && (.@yagu100 != .@yagu1) && (.@yagu10 != .@yagu1)) {
			// Store the actual number (unofficial).
			set .@yagu, (100*.@yagu100) + (10*.@yagu10) + .@yagu1;
			break;
		}
	}

	// The official time calculation is based on ServerTime,
	// an int HHMMSS equivalent to atoi(gettimestr("%H%M%S",7)).
	// UNIX time is really a better way to do this.
	set .@start_time, gettimetick(2);

	while(1) {

		// Guess a number that meets all conditions.
		while(1) {
			if (input(.@input,100,999) != 0) {
				mes "[ニア]";
				mes "数字合わせゲームを";
				mes "キャンセルしました。";
				close;
			}
			set .@input100, .@input / 100;
			set .@input10, (.@input % 100) / 10;
			set .@input1, .@input % 10;
			if ((.@input100 > 0) && (.@input10 > 0) && (.@input1 > 0)) {
				if ((.@input100 != .@input10) && (.@input100 != .@input1) && (.@input10 != .@input1))
					break;
			}
			mes "[ニア]";
			mes "0を含めて入力したか、";
			mes "同じ数字を複数の桁で";
			mes "使って入力しましたね。";
			mes "もう一度、別の数字を";
			mes "入力してください。";
			next;
		}

		set .@retry, .@retry+1;
		mes "[ニア]";
		mes "^ff0000"+strcharinfo(0)+"^000000さん、";
		mes "あなたの予想は^0000ff"+.@input+"^000000ですね。";
		mes "ちょっと";
		mes "結果を確認しますね。";
		next;

		// Correct digits in correct locations.
		set .@strike,0;
		if (.@yagu100 == .@input100)
			set .@strike, .@strike+1;
		if (.@yagu10 == .@input10)
			set .@strike, .@strike+1;
		if (.@yagu1 == .@input1)
			set .@strike, .@strike+1;

		// Correct digits in incorrect locations.
		set .@ball,0;
		if ((.@yagu100 == .@input10) || (.@yagu100 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu10 == .@input100) || (.@yagu10 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu1 == .@input100) || (.@yagu1 == .@input10))
			set .@ball, .@ball+1;

		if (.@strike == 3) {
			if (.@practice) {
				mes "[ニア]";
				mes "正解です！私が考えた";
				mes "数字は^ff0000"+.@yagu+"^000000でした！";
				mes "おめでとうございます、";
				mes "この練習ゲームに";
				mes "勝ちましたね！これで";
				mes "本番の準備もばっちりです～";
				close;
			} else {

				// Calculate final time and check against the previous record.
				set .@end_time, gettimetick(2);
				if (.@end_time < .@start_time)
					set .@end_time, .@end_time + 3600;
				set .@total_time, .@end_time - .@start_time;
				set .@total_min, .@total_time / 60;
				set .@total_sec, .@total_time % 60;
				set .@name_record$, $050908_minus1_yagu$;
				if (.@name_record$ == "") //custom
					set .@name_record$,"(未登録)";
				set .@score_record, $050908_minus1_yagu;
				set .@score_min, .@score_record / 60;
				set .@score_sec, .@score_record % 60;

				if (.@score_record >= .@total_time || !.@score_record) { //custom (second condition)
					set $050908_minus1_yagu$, strcharinfo(0);
					set $050908_minus1_yagu, .@total_time;
					mes "[ニア]";
					mes "やりました！";
					mes "私が考えた数字は";
					mes "^ff0000" + .@yagu + "^000000でした。タイムは";
					mes "^ff0000" + .@total_min + "分" + .@total_sec + "秒^000000、";
					mes "これまでの記録";
					mes "は^ff0000" + .@score_min + "分" + .@score_sec + "秒^000000でした。";
					next;
					mes "[ニア]";
					mes "^ff0000" + .@name_record$ + "^000000さんが";
					mes "持っていた記録でしたが、";
					mes "今度はあなたが";
					mes "数字合わせゲームの";
					mes "新しい目標です。";
					mes "おめでとうございます、新記録達成です！";
					close;
				}
				mes "[ニア]";
				mes "よくやりました、";
				mes "数字を当てましたね！答えは";
				mes "^ff0000" + .@yagu + "^000000でした。あなたは";
				mes "この数字を、正解として";
				mes "^ff0000" + .@total_min + "分" + .@total_sec + "秒^000000で導き出しましたね。";
				next;
				mes "[ニア]";
				mes "^ff0000" + .@name_record$ + "^000000さんが";
				mes "現在の数字合わせ";
				mes "ゲームの記録保持者です。タイムは";
				mes "^ff0000" + .@score_min + "分" + .@score_sec + "秒^000000です。";
				mes "いつかこの記録を";
				mes "破れるといいですね～";
				close;
			}
		} else {
			if (.@practice) {
				mes "[ニア]";
				mes "あなたの予想には^ff0000" + .@ball + "^000000個の";
				mes "正しい数字が含まれています。^ff0000" + .@strike + "^000000個は";
				mes "正しい桁に";
				mes "ありました。もう一度、";
				mes "挑戦してみて";
				mes "ください～";
				next;
			} else {
				mes "[ニア]";
				mes "今回の予想には^ff0000" + .@ball + "^000000個の";
				mes "正しい数字があり、";
				mes "^ff0000" + .@strike + "^000000個は";
				mes "正しい桁に";
				mes "入っていました。";
				switch(.@retry) {
					case 1: mes "残り4回です。"; break;
					case 2: mes "残り3回です。"; break;
					case 3: mes "残り2回です。"; break;
					case 4: mes "残り1回だけです…"; break;
					case 5: mes "答えは^ff0000" + .@yagu + "^000000でした。"; close;
				}
				next;
			}
		}
	}
}
