// Edgar's Offer :: iz_npc
//============================================================
-	script	::Edgar_izlude	-1,{
	if (MISC_QUEST & 16) {
		mes "[エドガー]";
		mes "またアルベルタへ向かうのか？前回と同じ割引をしてやろう、250 Zenyでいいぞ。";
		mes "どうだ、悪くないだろう？";
		next;
		if (select("よし、分かった～！:なんでそんなに優しくしてくれるんだ？！") == 1) {
			if (Zeny < 250) {
				mes "[エドガー]";
				mes "うーん…";
				mes "それじゃ";
				mes "足りないな。";
				mes "もう少し稼いでから";
				mes "また来てくれよ？";
				close ;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[エドガー]";
		mes "俺はこういう性分なんだよ。それに、お前のいたずらっぽい笑顔を見ていると、何年も前に船乗りになるといって家を出て行った、愛する金髪の息子を思い出すんだ。どこにいようと、あいつに幸あれ。";
		close;
	}
	mes "[エドガー]";
	mes "俺の町イズルードは、西の港でアルベルタと繋がっている。行き来があまりに多くて、もう半分アルベルタ人みたいなものさ。ヘヘヘ～";
	next;
	mes "[エドガー]";
	mes "アルベルタに住む、フェリックスってよく知ってる奴がいるんだが。あいつは本当にケチでな…何でもかんでも金を取るんだ！";
	next;
	mes "[エドガー]";
	mes "でも実はいい奴で、人助けが好きなんだ。心根は優しくて、あいつの言い値さえ払えば力を貸してくれるぞ。";
	next;
	mes "[エドガー]";
	mes "最近アルベルタでは、あいつは本当に人助けをしたいだけで、ジェロピーを要求するのはただの建前だって噂だ。まあ、一度あいつが何を持っているか見てみるといい。";
	next;
	if (select("アルベルタへの行き方を教えてくれますか？:会話を終える") == 1) {
		mes "[エドガー]";
		mes "ん？そうだな、自分の足で歩いていくこともできるぞ。だが金があるなら、船に乗ることをお勧めするね。";
		next;
		if (select("分かった。:でも歩くのはうんざりだし、金もないんだ！") == 1) {
			mes "[エドガー]";
			mes "よし、気をつけてな～";
			close;
		}
		mes "[エドガー]";
		mes "はあ…";
		mes "歩きたくない、金もない、それでも行きたいだと？やれやれ…";
		next;
		mes "[エドガー]";
		mes "分かった分かった。俺は船長だからな、安く連れて行ってやれるぞ。250 Zenyでどうだ？";
		next;
		if (select("よし、分かった～！:はあ、ぼったくりだ！！") == 1) {
			set MISC_QUEST,MISC_QUEST | 16;
			if (Zeny < 250) {
				mes "[エドガー]";
				mes "うーん…";
				mes "それじゃ";
				mes "足りないな。";
				mes "もう少し";
				mes "稼いでこい。";
				close;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[エドガー]";
		mes "やれやれ、";
		mes "それをぼったくりだと";
		mes "思うならな…";
		close;
	}
	mes "[エドガー]";
	mes "ああ、分かったよ。";
	mes "またな～";
	close;
}
