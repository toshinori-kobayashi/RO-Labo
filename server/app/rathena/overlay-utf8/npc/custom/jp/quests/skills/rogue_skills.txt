function	script	F_Kienna	{
	if (ROG_SK == 6) {
		donpcevent "#1stmove::OnDisable";
		mes "[キエナ]";
		mes "それでは、この訓練では";
		mes "私がどちらへ動くかを";
		mes "予測して、私の動きを";
		mes "塞いでもらいます。つまり";
		mes "私を身動き取れなくするんです。";
		mes "準備はいいですか～";
		set ROG_SK,7;
		next;
		for(set .@lim_1,0; .@lim_1<10; set .@lim_1,.@lim_1+1) {
			set .@move_1,rand(1,3);
			if (select("左を塞ぐ:右を塞ぐ:退路を塞ぐ") == .@move_1) {
				mes "[キエナ]";
				mes "あら…？";
				mes "塞がれちゃいました！";
				mes "とても上手です～";
				specialeffect EF_POTION1;
				specialeffect2 EF_POTION7;
				set .@suc_1,.@suc_1 + 1;
			}
			else {
				mes "[キエナ]";
				mes "残念ですが、";
				mes "その方向には動いていませんでした。";
				mes "塞ぐのは失敗です…";
			}
			next;
		}
		mes "[キエナ]";
		mes "はい、これで";
		mes "終わりです。あなたの成績は";
		switch(.@suc_1) { 
		case 10:
			mes "「^0000FFS^000000」でした。";
			mes "満点です！";
			break;
		case 9:
			mes "「^0000FFA^000000」でした。";
			mes "本当に上手ですね！";
			break;
		case 8:
			mes "「^0000FFB +^000000」でした。";
			mes "とても立派です！";
			break;
		case 7:
			mes "「^0000FFB^000000」でした。";
			mes "とても良い出来です～";
			break;
		case 6:
			mes "「^0000FFC +^000000」でした。";
			mes "悪くないですが、";
			mes "ギリギリの合格ですね！";
			break;
		case 5:
			mes "「^FF0000C^000000」でした。";
			break;
		case 4:
			mes "「^FF0000D+^000000」でした。";
			break;
		case 3:
			mes "「^FF0000D^000000」でした。";
			break;
		case 2:
			mes "「^FF0000F^000000」でした。";
			break;
		case 1:
			mes "「^FF0000F -^000000」でした。";
			break;
		case 0:
			mes "成績は…";
			mes "実は計算できません。";
			mes "一体";
			mes "何が起きたんでしょう！？";
		}
		next;
		if (.@suc_1 > 5) {
			mes "[キエナ]";
			mes "嬉しいことに、";
			mes "訓練は";
			mes "終了です！それでは";
			mes "トール・グレッグのもとへ";
			mes "戻して差し上げます、";
			mes "^FF0000クローズコンファイン^000000を最後まで習得してくださいね。";
			set ROG_SK,8;
			disablenpc "Kienna#"+getarg(0);
			donpcevent "#1st5min::OnDisable";
			enablenpc "#1strecog";
			close2;
			donpcevent "Waiting Room#rogue10::OnEnable";
			warp "in_rogue",264,124;
			end;
		}
		else {
			mes "[キエナ]";
			mes "うーん。この成績だと、";
			mes "まだクローズコンファインの";
			mes "習得を終えるには";
			mes "早いと思います。";
			mes "もう一度訓練を";
			mes "やり直しますか？";
			next;
			if (select("はい！:いいえ、結構です。") == 1) {
				disablenpc "Kienna#"+getarg(0);
				donpcevent "#1stmove::OnEnable";
				set ROG_SK,6;
				warp "prt_are01",150,150;
				end;
			}
			disablenpc "Kienna#"+getarg(0);
			donpcevent "#1st5min::OnDisable";
			enablenpc "#1strecog";
			set ROG_SK,6;
			donpcevent "Waiting Room#rogue10::OnEnable";
			warp "in_rogue",264,124;
			end;
		}
	}
	else if (ROG_SK == 7) {
		mes "[キエナ]";
		mes "訓練の途中で";
		mes "中断してしまったようですね。";
		mes "申し訳ありませんが、";
		mes "この訓練は";
		mes "最初からやり直して";
		mes "いただくことになります。";
		set ROG_SK,6;
		close2;
		disablenpc "Kienna#"+getarg(0);
		donpcevent "#1stmove::OnEnable";
		warp "prt_are01",150,150;
		end;
	}
	else if (ROG_SK == 8) {
		mes "[キエナ]";
		mes "クローズコンファインの";
		mes "訓練は";
		mes "もう終わっていますよ。";
		mes "もう私の手助けは";
		mes "必要ありません。";
		close2;
		donpcevent "Waiting Room#rogue10::OnEnable";
		warp "in_rogue",264,124;
		end;

		mes "[キエナ]";
		mes "申し訳ありませんが、";
		mes "ここに用はないようですね。";
		mes "迷っているなら";
		mes "ローグギルドへ送りましょう…";
		close2;
		donpcevent "Waiting Room#rogue10::OnEnable";
