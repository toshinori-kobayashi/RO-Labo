		mes "望むなら、今すぐ";
		mes "「グリード」を";
		mes "教え始めてやってもいいぞ。";
		mes "どうだ、坊主？";
		next;
		if (select("いいね！:また今度で～") == 1) {
			mes "[グッドマン]";
			mes "ふっ！いい選択だ。さて、";
			mes "「グリード」は使用者を中心とした";
			mes "5×5マスの範囲のアイテムを";
			mes "自動で拾い集めるスキルだ。";
			mes "モンスター討伐で散らばった品を拾うのに便利だぞ。";
			next;
			mes "[グッドマン]";
			mes "まずは第一段階だ。あんたが";
			mes "このスキルを覚える資格があるか";
			mes "試させてもらう。所持品を隙間なく";
			mes "アイテムで満たして、";
			mes "重量500の品が入らないほどにしてこい。";
			next;
			mes "[グッドマン]";
			mes "それが終わったら";
			mes "戻ってきて";
			mes "俺に話しかけろ。それじゃ";
			mes "がんばれよ～";
			BLACK_SK2 = 1;
			close;
		}
		mes "[グッドマン]";
		mes "そうか。他のことで";
		mes "忙しいのもいいが、";
		mes "時間ってのは貴重なもんだって";
		mes "忘れるなよ。いつだって";
		mes "一番価値のあることに";
		mes "使うべきだ。わかったか？";
		close;
	}
	if (BLACK_SK2 == 1) {
		if (MaxWeight - Weight >= 5000) {
			mes "[グッドマン]";
			mes "ふむ。戻ってくるのが";
			mes "早すぎるな。それに";
			mes "俺が出した課題を";
			mes "まだ終えてないだろう。よく聞け。";
			next;
			mes "[グッドマン]";
			mes "所持品を何でもいいから";
			mes "アイテムで満たして、";
			mes "重量500の品が";
			mes "入らないくらいいっぱいにしてこい。";
			mes "それが終わったら";
			mes "戻ってきて報告しろ。";
			close;
		}
		mes "[グッドマン]";
		mes "もう戻ってきたのか？";
		mes "よし、持ち物を";
		mes "確認させてもらうぞ。";
		mes "ええと、所持品、所持品っと…";
		next;
		mes "[グッドマン]";
		mes "よくやった、合格だ。";
		mes "これらのアイテムを集めるのに";
		mes "費やした時間と経験は、";
		mes "Zenyの値段以上の価値があるって";
		mes "わかっただろう？";
		next;
		mes "[グッドマン]";
		if (Upper == 1)
			mes "マスタースミスというのは";
		else
			mes "ブラックスミスというのは";
		mes "何よりもまず職人であるべきだと";
		mes "俺は思う。金持ちで有名になるのは";
		mes "単なる副産物にすぎない。";
		if (Upper == 1)
			mes "誇り高きマスタースミスは、価値は職人技にこそあると知っている…";
		else
			mes "誇り高きブラックスミスは、価値は職人技にこそあると知っている…";
		next;
		if (Upper == 1) {
			mes "[グッドマン]";
			mes "アイテムのZeny価値ってやつに";
			mes "あまりこだわらなければ、";
			mes "もっと簡単に";
			mes "手放せるようになる。";
			mes "俺がやってみせよう。";
			mes "ほら、これであんたにもできるはずだ。";
			next;
		}
		mes "[グッドマン]";
		mes "あんたが覚えたスキルは";
		mes "「グリード（強欲）」って名前だが、";
		mes "その名前とは違う使い方を";
		mes "してくれたらと思う。";
		mes "それを忘れずに、";
		mes "本物の傑作を作り上げてくれ。";
		skill "BS_GREED",1,SKILL_PERM;
		if (Upper == 1)
			BLACK_SK2 = 4;	// note: official script is using another variable when a transcendent character is doing the quest for the first time
		else
			BLACK_SK2 = 2;
		close;
	}
	mes "[グッドマン]";
	mes "「グリード」は";
	mes "うまく使えてるか？";
	mes "仲間たちのために";
	mes "もっと質のいい品を";
	mes "鍛造する助けに";
	mes "なっていればいいんだがな。";
	if (getskilllv("BS_GREED") == 0)	// debug
		skill "BS_GREED",1,SKILL_PERM;
	close;
}
