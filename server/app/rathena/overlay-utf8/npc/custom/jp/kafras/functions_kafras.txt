//=== Main Function ========================================================
//=   arg(0): Used to determine which welcome message to show.
//=   arg(1): Used to determine which menu to display.
//=   arg(2): Used to determine if the info menu is shown in F_KafInfo.
//=   arg(3): Cost to use Storage
//=   arg(4): Cost to Rent a Pushcart
//==========================================================================
function	script	F_Kafra	{
	callfunc "F_ClearGarbage"; // Clear outdated, unused variables
	.@welcome = getarg(0);
	.@menu_num = getarg(1);

	// Display Kafra Welcome Message
	switch(.@welcome) {
	default:// Default message (obsolete)
		mes "[カプラ職員]";
		mes "ようこそ";
		mes "カプラコーポレーションへ。";
		mes "カプラのサービスは";
		mes "常にあなたの味方です。";
		mes "本日はどのようなご用件でしょうか？";
		break;
	case 1:// Niflheim specific message
		mes "[カプラ職員]";
		mes "^666666よ、よ、ようこそ";
		mes "こ、ここはカ、カプラ";
		mes "しょ、しょうかいです…^000000";
		break;
	case 2:// Guild Castle Kafra message
		mes "[カプラ職員]";
		mes "ようこそ、^ff0000" + getguildname( getcharid(2) ) + "^000000のメンバー様。";
		mes "カプラコーポレーションは、どこへ行かれても共にございます。";
		break;
	case 3:// Amatsu specific message (obsolete)
		mes "[カプラ職員]";
		mes "遠い国から文化を学びに、それとも観光にいらしたのですか？";
		mes "どちらにしても、少しゆっくりしていかれませんか？";
		mes "この地にはいつも";
		mes "野花の香りが漂っています。";
		break;
	case 4:// Luoyang and Ayothaya specific message (obsolete)
		mes "[カプラ職員]";
		mes "各地にあるカプラの";
		mes "サービス拠点のおかげで、";
		mes "いつでも故郷は近くです。";
		break;
	case 5:// NPC has it's own welcome message. (Display nothing)
		break;
	}
	next;
	if (.@welcome == 2)// Guild Kafra Menu override (free Teleport, free Storage, Free Cart)
		setarray .@K_Menu0$[0], "倉庫を利用する", "ギルド倉庫を利用する", "カートをレンタルする", "テレポートサービス", "キャンセル";
	else {
		switch(.@menu_num) {
			// Save and Storage only
			case 1:	setarray .@K_Menu0$[0],"セーブ","倉庫を利用する","キャンセル"; break;
			// Storage only
			case 2:	setarray .@K_Menu0$[0],"倉庫を利用する","キャンセル"; break;
			// No Teleport (Common)
			case 3: setarray .@K_Menu0$[0],"セーブ","倉庫を利用する","カートをレンタルする","その他の情報","キャンセル"; break;
			// Case 4 is Einbroch no tele message.
			// No save, or teleport. (Common)
			case 5:	setarray .@K_Menu0$[0],"倉庫を利用する","カートをレンタルする","その他の情報","キャンセル"; break;
			// Storage and Check Other Information only.
			case 6:	setarray .@K_Menu0$[0],"倉庫を利用する","その他の情報","キャンセル"; break;
			// Save, Storage, and Pushcart only (Kafra Warehouse)
			case 7:	setarray .@K_Menu0$[0],"セーブ","倉庫を利用する","カートをレンタルする","キャンセル"; break;
			// Save, Storage, Other Check information. (Turbo track)
			case 8: setarray .@K_Menu0$[0],"セーブ","倉庫を利用する","その他の情報","キャンセル"; break;
			// No Save (Rune Knight)
			case 9: setarray .@K_Menu0$[0],"倉庫を利用する","カートをレンタルする","テレポートサービス","その他の情報","キャンセル"; break;
			// Storage, Save, and Pushcart (Dewata, reorder of case 7)
			case 10: setarray .@K_Menu0$[0],"倉庫を利用する","セーブ","カートをレンタルする","キャンセル"; break;
			// Default message (obsolete)
			default: setarray .@K_Menu0$[0],"セーブ","倉庫を利用する","テレポートサービス","カートをレンタルする","その他の情報","キャンセル"; break;
		}
	}
	.@menu$ = implode(.@K_Menu0$,":");
	while (true) {
		.@j = select(.@menu$) - 1;
		if (.@K_Menu0$[.@j] == "セーブ")
			return;
		else if (.@K_Menu0$[.@j] == "倉庫を利用する") {
			if (.@welcome == 2)// Do not charge for Guild Storage
				callfunc "F_KafStor",2,0,0;
			else
				callfunc "F_KafStor",0,getarg(3),.@welcome;
			next;
		}
		else if (.@K_Menu0$[.@j] == "テレポートサービス") {
			if (.@menu_num != 4)// Display Einbroch "No Teleport Service" notice.
				callfunc "F_KafTele",.@welcome;
			else {
				mes "[カプラ職員]";
				mes "シュヴァルツヴァルト共和国では、";
				mes "^FF0000輸送制限協定^000000により、";
				mes "カプラコーポレーションは";
				mes "テレポートサービスを";
				mes "提供できません。";
				next;
				mes "[カプラ職員]";
				mes "お手数ですが";
				mes "飛行船サービスを";
				mes "ご利用ください。ご理解と";
				mes "ご協力に感謝いたします。";
			}
			next;
		}
		else if (.@K_Menu0$[.@j] == "カートをレンタルする") {
			if (callfunc("F_KafCart",.@welcome,getarg(4)) == 1)
				next;
		}
		else if (.@K_Menu0$[.@j] == "その他の情報") {
			callfunc "F_KafInfo",getarg(2);
			next;
		}
		else if (.@K_Menu0$[.@j] == "キャンセル") {
			callfunc "F_KafEnd",.@welcome,0;
			end;
		}
		else if (.@K_Menu0$[.@j] == "ギルド倉庫を利用する") {
			callfunc "F_KafStor",1,0;
			next;
		}
	}
}

//===================================================
// Storage Function
//===================================================
function	script	F_KafStor	{
	.@type = getarg(0);
	.@fee = getarg(1);

	// Unable to access Guild Storage (Busy)
	if (.@type == 1) {
		if (guildopenstorage()) {
			mes "[カプラ職員]";
			mes "只今、他のギルドメンバーが倉庫を";
			mes "利用中です。少々お待ちください。";
			close3;
		}
		cutin "", 255;
		close;
	}
	// Unable to access Normal Storage (Insufficient Basic Skills)
	if (!callfunc("F_CanOpenStorage")) {
		mes "[カプラ職員]";
		// Niflheim Specific Message
		if (getarg(2) == 1) {
			mes "^666666す、すみません";
			mes "あ、あなたはノ、ノービス";
			mes "き、基礎スキルレベル";
			mes "6が、ひ、必要です…^000000";
			return;
		}
		mes "申し訳ございませんが、";
		mes "ノービスの";
		mes "基礎スキルレベル6が";
		mes "倉庫サービスの利用に必要です。";
		return;
	}
	// Accessing Normal Storage (Skipped if accessing Storage from Guild castle)
	if (.@type != 2) {
		// Consume "Free Ticket for Kafra Storage" if available.
		if (countitem(7059))
			delitem 7059,1;
		else {
			if (Zeny < .@fee) {
				mes "[カプラ職員]";
				// Niflheim Specific Message
				if (getarg(2) == 1) {
					percentheal -50,-50;
					mes "^666666Z、Zeny…";
					mes "も、もっとZeny…！";
					mes "150…Zeny、ひ、必要…";
					mes "うっ！ち、血を…！^000000";
					return;
				}
				// Standard Message
				mes "申し訳ございませんが、";
				mes "Zenyが不足しており、";
				mes "倉庫サービスをご利用いただけません。";
				mes "倉庫の利用料金は" + .@fee + "Zenyです。";
				return;
			}
			Zeny -= .@fee;
			RESRVPTS += (.@fee/5);
		}
	}

	// Surpress the "End" message if in guild castle.
	if (.@type != 2) {
		// Niflheim specific Message
		if (getarg(2) == 1) {
			percentheal 0,-10;
			mes "[カプラ職員]";
			for ( .@i = 0; .@i < 5; .@i++ )
				mes "^666666あ、ありがとう…ございました…^000000";
		}
		// Normal message
		else {
			mes "[カプラ職員]";
			mes "それでは";
			mes "倉庫をお開けします。";
			mes "ご利用いただき";
			mes "ありがとうございました。";
		}
	}
	callfunc("F_CheckKafCode");	//check your storage password, if set
	close2;
	openstorage;
	cutin "", 255;
	end;
}

//===================================================
// Teleport Function
//===================================================
function	script	F_KafTele	{
	mes "[カプラ職員]";
	mes "行き先を";
	mes "お選びください。";
	next;
	.@j = select( implode(@wrpC$,":") ) - 1;
	if (@wrpC$[.@j] == "キャンセル")
		return;
	// Consume "Free Ticket for Kafra Transportation" if available.
	// Do not consume if in Guild castle
	if (countitem(7060) > 0 && getarg(0) != 2)
		delitem 7060,1;
	else {
		if (Zeny < @wrpP[.@j]) {
			mes "[カプラ職員]";
			mes "申し訳ございませんが、";
			mes "テレポートサービスに必要な";
			mes "Zenyが不足しています。";
			mes "" + @wrpD$[.@j] + "までの料金は" + @wrpP[.@j] + "Zenyです。";
			close3;
		}
		Zeny -= @wrpP[.@j];
		RESRVPTS += (@wrpP[.@j]/16);
	}
	cutin "", 255;
	if (@wrpD$[.@j] == "アルデバラン") warp "aldebaran",168,112;
	else if (@wrpD$[.@j] == "アルベルタ") warp "alberta", 117, 56;
	else if (@wrpD$[.@j] == "コモド") warp "comodo", 209, 143;
	else if (@wrpD$[.@j] == "コモド灯台") warp "cmd_fild07", 127, 134;
	else if (@wrpD$[.@j] == "ゲフェン") warp "geffen", 120, 39;
	else if (@wrpD$[.@j] == "イズルード") {
		if (checkre(0) == true)
			warp "izlude", 128, 98;
		else
			warp "izlude", 91, 105;
	}
	else if (@wrpD$[.@j] == "ジュノー") warp "yuno", 158, 125;
	else if (@wrpD$[.@j] == "ミョルニル死の穴") warp "mjolnir_02", 99, 351;
	else if (@wrpD$[.@j] == "モロク") warp "morocc", 156, 46;
	else if (@wrpD$[.@j] == "オークダンジョン") warp "gef_fild10", 52, 326;
	else if (@wrpD$[.@j] == "フェイヨン") warp "payon", 161, 58;
	else if (@wrpD$[.@j] == "プロンテラ") warp "prontera", 116, 72;
	else if (@wrpD$[.@j] == "ウンバラ") warp "umbala", 100, 154;
	end;
}


//===================================================
// Cart Function
//===================================================
function	script	F_KafCart	{
	// Ensure that the class wanting to rent a pushcart is a merchant
	if (BaseClass != Job_Merchant) {
		mes "[カプラ職員]";
		mes "申し訳ございませんが、";
		mes "カートレンタルサービスは";
		if (checkre(0) == true)
			mes "商人系職業限定となります。";
		else {
			mes "商人、ブラックスミス、";
			mes "マスタースミス、";
			mes "アルケミスト、バイオケミスト限定です。";
		}
		return 1;
	}
	// Make sure the invoking character does not have a cart already
	else if (checkcart() == true) {
		mes "[カプラ職員]";
		mes "すでにカートを";
		mes "装備されています。";
		mes "申し訳ございませんが、";
		mes "お一人様一台までしか";
		mes "レンタルできません。";
		return 1;
	}
	// Check if the player has the Pushcart skill
	else if (getskilllv("MC_PUSHCART") == 0) {
		mes "[カプラ職員]";
		mes "「カート引き」スキルを習得すると、カートをレンタルできます。";
		return 1;
	}
	// Consume "Free Ticket for the Cart Service" if available.
	// Do not consume if in Guild castle.
	if (countitem(7061) > 0 && getarg(0) != 2)
		delitem 7061,1;
	else {
		.@rental_fee = getarg(1);
		mes "[カプラ職員]";
		mes "カートのレンタル料金は";
		mes "" + .@rental_fee + "Zenyです。レンタル";
		mes "なさいますか？";
		next;
		if (select("カートをレンタルする。:キャンセル") == 2)
			return 0;
		if (Zeny < .@rental_fee) {
			mes "[カプラ職員]";
			mes "申し訳ございませんが、";
			mes "Zenyが不足しているため、";
			mes "カートをレンタル";
			mes "できません。料金は" + .@rental_fee + "Zenyです。";
			return 1;
		}
		Zeny -= .@rental_fee;
		RESRVPTS += 48;
	}
	setcart;
	return 1;
}

function	script	F_KafInfo	{
// Uncomment next line to block Kafra Storage Protection

	if (getarg(0) == 2)// Port Malaya, Rune Knight
		setarray .@menu$[0], "特別予約ポイントを確認する。", "", "", "キャンセル";

	while (true) {
		.@j = select(.@menu_list$) - 1;
		switch(.@j) {
		case 0:// Check Special Reserve Points.
			mes "[カプラ職員]";
			mes "ええと…";
			mes "" + strcharinfo(0) + "...";
			mes "現在、あなたは合計で";
			mes "" + RESRVPTS + "特別予約ポイントをお持ちです。";
			next;
			mes "[カプラ職員]";
			mes "特別予約ポイントは";
			mes "アルデバランのカプラ本社で";
			mes "景品と交換できます。";
			mes "ぜひサービスをご活用ください。";
			if (getarg(0) == 1)
				return;
			next;
			break;
		case 1:// Storage Password Service
			callfunc("F_SetKafCode","[カプラ職員]","カプラサービス");
		case 2:// Kafra Employee Locations
			for ( .@i = 0; .@i <= 3; .@i++ )
				viewpoint 1,@viewpX[.@i],@viewpY[.@i],(.@i+1),0xFF00FF;
			next;
			for ( .@i = 0; .@i <= 3; .@i++ )
				viewpoint 2,@viewpX[.@i],@viewpY[.@i],(.@i+1),0xFF00FF;
			break;
		default:
			return;
		}
	}
}

function	script	F_KafEnd	{
	.@type = getarg(0);
	mes "[カプラ職員]";
	if (getarg(1) == true) {// Save
		mes "あなたのリスポーンポイントを";
		mes "以下の場所に保存しました：";
		mes "" + getarg(2) + "。";
		mes "ご利用いただき";
		mes "ありがとうございました。";
	} else if (.@type == 0 || .@type == 5) {// Generic End
		mes "私たちカプラコーポレーションは、";
		mes "常に最高のサービス提供に努めております。";
	} else if (.@type == 1) {// Niflheim End
		percentheal 0,-25;
		mes "^666666カ、カプラは";
		mes "け、決して死な、ない…";
		mes "あ、あなたの味方です～^000000";
	} else if (.@type == 2) {// RockRidge Save
		mes "セーブしました。";
		mes "ご利用ありがとうございました。";
	}
	close3;
}

//===================================================
// Check Storage Password Function
//===================================================
function	script	F_CheckKafCode	{
	if (!#kafra_code)
		return;
	mes "保管庫のパスワードを入力してください：";
	input .@code_;
	if (.@code_ != #kafra_code) {
		dispbottom "保管庫のパスワードが違います。";
		close3;
	}
	@kafcode_try = 0;
	return;
}

// === Set / Change / Clear Storage Password Function ===
// getarg(0): NPC Name
// getarg(1): Company Name
//=======================================================
function	script	F_SetKafCode	{
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	if (#kafra_code == 0) {
		mes "" + .@comp_name$ + "は新しいサービスをご案内いたします：";
		mes "パスワードによる倉庫の追加保護サービスです。";
		next;
		if (select( "新しいパスワードを設定 -> 5000z", "キャンセル" ) == 1)
			callsub( S_SET, .@npc_name$, .@comp_name$ );
		close3;
	}
	mes "倉庫はパスワードで保護中です。どうしますか？";
	next;
	switch( select( "古いパスワードを変更 -> 5000z", "倉庫パスワードを解除 -> 1000z", "キャンセル" ) ) {
	case 1:
		mes .@npc_name$;
		mes "まず、^0000FF現在のパスワード^000000を入力してください。";
		.@code = callfunc("F_EntKafCode");
		if (!.@code || .@code != #kafra_code) {
			mes "パスワードが違います。新しいパスワードは設定できません。";
			emotion ET_SCRATCH;
			break;
		}
		next;
		callsub( S_SET, .@npc_name$, .@comp_name$ );
		break;
	case 2:
		mes .@npc_name$;
		mes "解除する前に、現在のパスワードを入力してください。";
		.@code = callfunc("F_EntKafCode");
		if (!.@code) {
			mes "パスワードは解除されませんでした。";
			emotion ET_SCRATCH;
			break;
		}
		next;
		mes .@npc_name$;
		if (Zeny < 1000) {
			mes "Zenyが不足しています。";
			emotion ET_MONEY;
			break;
		}
		Zeny -= 1000;
		if (.@code == #kafra_code) {
			#kafra_code = 0;
			mes "保管庫のパスワードを正常に解除しました。";
			mes "" + .@comp_name$ + "をご利用いただき、ありがとうございました。";
			emotion ET_THANKS;
		} else {
			mes "パスワードが違います。1000zは返却されません。";
			mes "次回は正しいパスワードを入力してください。";
			emotion ET_SORRY;
		}
		break;
	case 3:
		break;
	}
	close3;

S_SET:
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	mes "^FF0000新しいパスワード^000000を入力し、盗難から倉庫を守りましょう。";
	.@code = callfunc("F_EntKafCode");
	if (!.@code) {
		mes "パスワードは変更されませんでした。";
		emotion ET_SCRATCH;
		return;
	}
	next;
	mes .@npc_name$;
	if (Zeny < 5000) {
		mes "Zenyが不足しています。";
		emotion ET_MONEY;
		return;
	}
	Zeny -= 5000;
	#kafra_code = .@code;
	mes "秘密のパスワードで倉庫を保護しました。";
	mes "" + .@comp_name$ + "をご利用いただき、ありがとうございました。";
	emotion ET_THANKS;
	return;
}

//===================================================
// Basic Password Validation Function
//===================================================
function	script	F_EntKafCode	{
	mes "1000～10000000の数字を入力してください：";
	@kafcode_try += 1;
	if (@kafcode_try > 10) {
		@kafcode_try = 0;
		logmes "Tried to fit storage password.";
		return 0;
	}
	if (input(.@code_) == 1) {
		mes "そのパスワードは大きすぎます。";
		return 0;
	}
	if (.@code_ < 1000) {
		mes "そのパスワードは短すぎます。";
		return 0;
	}
	return .@code_;
}
