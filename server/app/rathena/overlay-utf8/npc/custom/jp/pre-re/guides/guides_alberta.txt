alberta,23,238,4	script	案内係#alb::AlbGuide	105,{
	cutin "prt_soldier",2;
	mes "[アルベルタ案内係]";
	mes "港町アルベルタへ";
	mes "ようこそ。街で";
	mes "何か困ったことが";
	mes "あれば、";
	mes "案内が必要なときも";
	mes "気軽に聞いてください。";
	set .@loop1,1;
	while (.@loop1) {
		next;
		switch(select("街の案内:ミニマップの印を消す:お知らせ:キャンセル")) {
		case 1:
			mes "[アルベルタ案内係]";
			mes "以下のメニューから";
			mes "行きたい場所を";
			mes "選んでください。";
			if (.@compass_check == 0) {
				mes "ミニマップに";
				mes "行き先の印を";
				mes "つけましょうか？";
				next;
				if (select("はい:いいえ") == 1) set .@compass_check,1;
			}
			set .@loop2,1;
			while (.@loop2) {
				if (.@wait_button_chk == 0) set .@wait_button_chk,1;
				else next;
				
				switch(select("^FF0000商人ギルド^000000:武器屋:道具屋:宿屋:鍛冶屋:キャンセル")) {
				case 1:
					mes "[アルベルタ案内係]";
					mes "商人ギルドは";
					mes "商人への転職を";
					mes "扱っています。";
					mes "アルベルタの";
					mes "南西の角に";
					mes "あります。";
					if (.@compass_check)
						viewpoint 1,33,41,2,0xFF0000;
					break;
				case 2:
					mes "[アルベルタ案内係]";
					mes "武器屋は";
					mes "アルベルタの";
					mes "南端にあります。";
					if (.@compass_check)
						viewpoint 1,117,37,3,0xFF00FF;
					break;
				case 3:
					mes "[アルベルタ案内係]";
					mes "道具屋は";
					mes "アルベルタの";
					mes "中心に近く、";
					mes "すぐに";
					mes "見つかります。";
					if (.@compass_check)
						viewpoint 1,98,154,4,0xFF00FF;
					break;
				case 4:
					mes "[アルベルタ案内係]";
					mes "宿屋は";
					mes "アルベルタの";
					mes "北端にあり、";
					mes "休むことができます。";
					if (.@compass_check)
						viewpoint 1,65,233,5,0xFF00FF;
					break;
				case 5:
					mes "[アルベルタ案内係]";
					mes "アルベルタの鍛冶屋は";
					mes "商人ギルドと";
					mes "同じ建物にあり、";
					mes "南西にあります。";
					if (.@compass_check)
						viewpoint 1,35,41,6,0xFF00FF;
					break;
				case 6:
					mes "[アルベルタ案内係]";
					mes "「ミニマップの印を";
					mes "消す」を選んで";
					mes "いただければ、";
					mes "地図上の印を";
					mes "消去できます。";
					set .@loop2,0;
				}
			}
			break;
		case 2:
			viewpoint 2,237,41,2,0xFF0000;
			viewpoint 2,237,41,3,0xFF00FF;
			viewpoint 2,46,345,4,0xFF00FF;
			viewpoint 2,175,220,5,0xFF00FF;
			viewpoint 2,175,220,6,0xFF00FF;
			set .@compass_check,0;
			break;
		case 3:
			mes "[アルベルタ案内係]";
			mes "魔法と技術の進歩により";
			mes "情報システムが";
			mes "更新されました。";
			mes "これにより、ミニマップに";
			mes "場所の印をつけて";
			mes "案内できるようになりました。";
			next;
			mes "[アルベルタ案内係]";
			mes "ミニマップは画面の";
			mes "右上にあります。";
			mes "見当たらない場合は";
			mes "Ctrl+Tabキーを押すか、";
			mes "基本情報の「Map」ボタンでも確認できます。";
			next;
			mes "[アルベルタ案内係]";
			mes "ミニマップ上の";
			mes "「+」「-」の記号で";
			mes "拡大・縮小できます。";
			mes "アルベルタでの";
			mes "冒険を";
			mes "楽しんでください。";
			break;
		case 4:
			mes "[アルベルタ案内係]";
			mes "旅をするときは";
			mes "気をつけて。";
			mes "何か分からないことが";
			mes "あれば、いつでも聞いてください。";
			close2;
			set .@loop1,0;
		}
	}
	cutin "prt_soldier",255;
	end;
}

alberta,120,60,3	duplicate(AlbGuide)	案内係#2alb::Guide#2alb	105
