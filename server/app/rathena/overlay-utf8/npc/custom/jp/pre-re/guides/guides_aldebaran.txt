aldebaran,139,63,4	script	衛兵#alde::Soldier#alde	105,{
	cutin "prt_soldier",2;
	mes "[アルデバラン衛兵]";
	mes "私はどこにでもいる";
	mes "ただの衛兵です。名前もないような者ですが…";
	next;
	mes "[アルデバラン衛兵]";
	mes "私はアルデバラン駐屯部隊の道案内係を務めています。";
	mes "この街を案内しましょう！";
	next;
	if (select("位置を案内してもらう:会話を終える") == 1) {
		viewpoint 1,61,229,0,0xFF6633;
		viewpoint 1,72,197,1,0x0000FF;
		viewpoint 1,223,222,2,0x00FFFF;
		viewpoint 1,233,105,3,0x515151;
		viewpoint 1,197,70,4,0x3355FF;
		viewpoint 1,60,60,5,0xFF5555;
		mes "^FF6633+^000000 -> カプラ本店";
		mes "^0000FF+^000000 -> 武器屋";
		mes "^00FFFF+^000000 -> 魔道士ギルド（閉鎖中）";
		mes "^515151+^000000 -> 酒場";
		mes "^3355FF+^000000 -> 道具屋";
		mes "^FF5555+^000000 -> アルケミストギルド";
		close2;
		cutin "prt_soldier",255;
		end;
	}
	mes "[アルデバラン衛兵]";
	mes "我々はアルデバランを守る誓いを立てた者。";
	mes "悪しき者には正義の鉄拳を！";
	close2;
	cutin "prt_soldier",255;
	end;
}

aldebaran,133,108,4	script	衛兵#2alde::Soldier#2alde	105,{

	switch(select("カプラ本店:武器屋:魔道士ギルド:酒場:道具屋:アルケミストギルド:会話を終える")) {

		break;

		break;

		mes "^515151+^000000 -> 酒場";
		break;
	case 5:

		break;

		viewpoint 1,60,60,5,0xFF5555;

	end;
}
