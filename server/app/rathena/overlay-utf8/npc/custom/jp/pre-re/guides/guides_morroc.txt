	cutin "moc_soldier",2;
	mes "[モロク案内係]";
	mes "モロクへようこそ。";
	mes "ルーンミッドガッツ王国の";
	mes "辺境の町です。";
	mes "街で何か困ったことが";
	mes "あれば、";
	mes "気軽に聞いてください。";
	set .@loop1,1;
	while (.@loop1) {
		next;
		switch(select("街の案内:ミニマップの印を消す:お知らせ:キャンセル")) {
		case 1:
			mes "[モロク案内係]";
			mes "以下のメニューから";
			mes "行きたい場所を";
			mes "選んでください。";
			if (.@compass_check == 0) {
				mes "ミニマップに";
				mes "行き先の印を";
				mes "つけましょうか？";
				next;
				if (select("はい:いいえ") == 1) set .@compass_check,1;
			}
			set .@loop2,1;
			while(.@loop2) {
				if (.@wait_button_chk == 0) set .@wait_button_chk,1;
				else next;
				
				switch(select("^FF0000シーフギルド^000000:武器屋:宿屋:酒場:傭兵ギルド:鍛冶屋:キャンセル")) {
				case 1:
					mes "[モロク案内係]";
					mes "シーフギルドは";
					mes "シーフへの転職を";
					mes "扱っています。";
					mes "噂によると、";
					mes "近くのピラミッドの中に";
					mes "あるらしいです…";
					if (.@compass_check)
						viewpoint 1,24,297,2,0xFF0000;
					break;
				case 2:
					mes "[モロク案内係]";
					mes "武器屋は";
					mes "モロクの";
					mes "南東端にあります。";
					if (.@compass_check)
						viewpoint 1,253,56,3,0xFF00FF;
					break;
				case 3:
					mes "[モロク案内係]";
					mes "宿屋は";
					mes "モロクの南東と";
					mes "北東の両方に";
					mes "あります。";
					if (.@compass_check) {
						viewpoint 1,197,66,4,0xFF00FF;
						viewpoint 1,273,269,5,0xFF00FF;
					}
					break;
				case 4:
					mes "[モロク案内係]";
					mes "酒場は";
					mes "モロク北東にあります。";
					if (.@compass_check)
						viewpoint 1,52,259,6,0xFF00FF;
					break;
				case 5:
					mes "[モロク案内係]";
					mes "傭兵ギルドは";
					mes "モロクの";
					mes "東側にあります。";
					if (.@compass_check)
						viewpoint 1,284,171,7,0x00FF00;
					break;
				case 6:
					mes "[モロク案内係]";
					mes "鍛冶屋は";
					mes "モロク中央から";
					mes "南西へ向かった";
					mes "ところにあります。";
					if (.@compass_check)
						viewpoint 1,47,47,7,0xFF00FF;
					break;
				case 7:
					mes "[モロク案内係]";
					mes "「ミニマップの印を";
					mes "消す」を選んで";
					mes "いただければ、";
					mes "地図上の印を";
					mes "消去できます。";
					set .@loop2,0;
				}
			}
			break;
		case 2:
			viewpoint 2,237,41,2,0x00FF00;
			viewpoint 2,237,41,3,0x0000FF;
			viewpoint 2,46,345,4,0x00FF00;
			viewpoint 2,175,220,5,0xFF0000;
			viewpoint 2,175,220,6,0xFF0000;
			viewpoint 2,175,220,7,0xFF0000;
			set .@compass_check,0;
			break;
		case 3:
			mes "[モロク案内係]";
			mes "魔法と技術の進歩により";
			mes "情報システムが";
			mes "更新されました。";
			mes "これにより、ミニマップに";
			mes "場所の印をつけて";
			mes "案内できるようになりました。";
			next;
			mes "[モロク案内係]";
			mes "ミニマップは画面の";
			mes "右上にあります。";
			mes "見当たらない場合は";
			mes "Ctrl+Tabキーを押すか、";
			mes "基本情報の「Map」ボタンでも確認できます。";
			next;
			mes "[モロク案内係]";
			mes "ミニマップ上の";
			mes "「+」「-」の記号で";
			mes "拡大・縮小できます。";
			mes "モロクでの";
			mes "冒険を";
			mes "楽しんでください。";
			break;
		case 4:
			mes "[モロク案内係]";
			mes "それじゃあ、";
			mes "あまり面倒事には";
			mes "巻き込まれないように";
			mes "気をつけてくださいね。";
			close2;
			set .@loop1,0;
		}
	}
	cutin "moc_soldier",255;
	end;
}
