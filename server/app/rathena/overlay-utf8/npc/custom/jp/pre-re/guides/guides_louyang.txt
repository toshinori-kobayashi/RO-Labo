			break;
		case 3:
			mes "[衛兵]";
			mes "役所では政府の仕事が行われている。";
			mes "場所は^A5BAAD+^000000だ。";
			next;
			mes "[衛兵]";
			mes "何か困りごとがあれば、";
			mes "役所の";
			mes "職員に相談するといい。";
			viewpoint 1,309,80,4,0xA5BAAD;
			break;
		case 4:
			mes "[衛兵]";
			mes "武器屋は^55FF33+^000000の場所にある。";
			next;
			mes "[衛兵]";
			mes "龍之城の";
			mes "熟練の鍛冶職人が鍛えた";
			mes "見事な武器の数々を";
			mes "目にすることができるぞ。";
			viewpoint 1,145,174,5,0x55FF33;
			break;
		case 5:
			mes "[衛兵]";
			mes "道具屋は^3355FF+^000000の場所にある。";
			next;
			mes "[衛兵]";
			mes "敵を知ることは";
			mes "戦いの半分を制するようなものだ！";
			mes "後で後悔するより、備えておく方が安全というものだろう。品揃えを見てきてはどうだ？";
			viewpoint 1,135,98,6,0x3355FF;
			break;
		case 6:
			mes "[衛兵]";
			mes "旅の途中で疲れたら、酒場に寄ることを勧める。場所は^00FF00+^000000だ。";
			next;
			mes "[衛兵]";
			mes "酒場は他の旅人と出会い、";
			mes "役に立つ話を聞くのにもってこいの場所だ。";
			viewpoint 1,280,167,7,0x00FF00;
			break;
		case 7:
			mes "[衛兵]";
			mes "ミニマップの印を";
			mes "全部消したければ、メニューから「ミニマップの印を全部消す」を選んでくれ。";
		}
		break;
	case 2:
		viewpoint 2,218,255,2,0xFF00FF;
		viewpoint 2,263,94,3,0xFF00FF;
		viewpoint 2,309,80,4,0xFF00FF;
		viewpoint 2,145,174,5,0xFF00FF;
		viewpoint 2,135,98,6,0xFF00FF;
		viewpoint 2,280,167,7,0xFF00FF;
		mes "[衛兵]";
		mes "よし、ミニマップの印は全部消したぞ。施設の場所が知りたくなったら、いつでも聞いてくれ。";
		break;
	case 3:
		mes "[衛兵]";
		mes "たまには自分で";
		mes "探検してみるのも";
		mes "楽しいものだろうな。気をつけてな。";
	}
	close;
}
