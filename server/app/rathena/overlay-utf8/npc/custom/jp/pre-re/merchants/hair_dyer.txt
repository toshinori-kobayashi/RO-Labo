prt_in,243,168,4	script	ジョヴォヴィッチ::Jovovich	91,{
	mes "[美容師ジョヴォヴィッチ]";
	mes "いらっしゃいませ～！";
	mes "どういったご用件でしょう？";
	next;
	mes "[美容師ジョヴォヴィッチ]";
	if (Sex == SEX_MALE) mes "あらあら！髪が傷んでいますね。プロの手入れが必要そうです。こちらへどうぞ、さあさあ。";
	else {
		if (rand(20) > 11) {
			mes "え！？あらまあ！だめだめだめ！髪がとっても傷んでいますよ！このままにしておくのは良くありません。";
			next;
			mes "[美容師ジョヴォヴィッチ]";
			mes "髪の手入れをさせていただけませんか？お願いします。";
		}
		else mes "わあ！髪を染めればもっと素敵になりますよ～気分転換に染めてみませんか？";
	}
	next;
	while(1) {
		switch(select("髪を染める:ヒントと情報:キャンセル")) {
		case 1:
			mes "[美容師ジョヴォヴィッチ]";
			mes "はいはい、良い選択ですね～";
			mes "さあ、イメチェンのチャンスです、美しく花開くチャンスですよ！お好きな色をお選びください。";
			next;
			while(1) {
				if (.@choose_success == 1) {
					mes "[美容師ジョヴォヴィッチ]";
					mes "いかがですか？別の色にしてみたいですか？";
					next;
					if (select("はい:いいえ") == 1) {
						mes "[美容師ジョヴォヴィッチ]";
						mes "わかりました！お好きな色をお選びください。";
						next;
					}
					else  {
						mes "[美容師ジョヴォヴィッチ]";
						mes "うーん、ちょっと残念です。もっと良い仕事をしたかったのですが。次はきっと頑張りますね。また来てください～";
						close;
					}
				}

				while(1) {
					switch(select("赤にしてください。:黄にしてください。:紫にしてください。:橙にしてください。:緑にしてください。:青にしてください。:白にしてください。:黒にしてください。:やっぱり、今の髪色のままでいいです。")) {
					case 1: set .@headpalette,8; break;
					case 2: set .@headpalette,1; break;
					case 3: set .@headpalette,2; break;
					case 4: set .@headpalette,3; break;
					case 5: set .@headpalette,4; break;
					case 6: set .@headpalette,5; break;
					case 7: set .@headpalette,6; break;
					case 8: set .@headpalette,7; break;
					case 9:
						if (.@choose_success != 0) {
							mes "[美容師ジョヴォヴィッチ]";
							mes "今の髪色がお気に入りなんですね～";
							close;
						}
						mes "[美容師ジョヴォヴィッチ]";
						mes "ええ～？髪を染めないんですか？ちょっと寂しいです…";
						close;
					}

					if (.@headpalette == getlook(VAR_HEADPALETTE)) {
						mes "[美容師ジョヴォヴィッチ]";
						mes "あら？それは今の髪色と同じですよ。別の色をお選びください。";
						next;
					}
					else {
						switch(.@headpalette) {
						case 1: callsub S_NoDye,976,"黄"; break;
						case 2: callsub S_NoDye,978,"紫"; break;
						case 3: callsub S_NoDye,980,"橙"; break;
						case 4: callsub S_NoDye,979,"緑"; break;
						case 5: callsub S_NoDye,981,"青"; break;
						case 6: callsub S_NoDye,982,"白"; break;
						case 7: callsub S_NoDye,983,"黒"; break;
						case 8: callsub S_NoDye,975,"赤"; break;
						}
						if (Zeny < 1000) {
							mes "[美容師ジョヴォヴィッチ]";
							mes "料金は1000Zenyです。足りないのでは…？";
							close;
						}

						switch(.@headpalette) {
						case 1: delitem 976,1; break; //Lemon_Dyestuffs
						case 2: delitem 978,1; break; //Violet_Dyestuffs
						case 3: delitem 980,1; break; //Orange_Dyestuffs
						case 4: delitem 979,1; break; //Darkgreen_Dyestuffs
						case 5: delitem 981,1; break; //Cobaltblue_Dyestuffs
						case 6: delitem 982,1; break; //White_Dyestuffs
						case 7: delitem 983,1; break; //Black_Dyestuffs
						case 8: delitem 975,1; break; //Scarlet_Dyestuffs
						}
						set Zeny, Zeny-1000;
						setlook VAR_HEADPALETTE,.@headpalette;
						set .@choose_success,1;
						break;
					}

				}

			}

		case 2:
			mes "[美容師ジョヴォヴィッチ]";
			mes "落ち込んでいる時、失恋した時、誰かに良い印象を与えたい時、それにただ目立ちたい時だって…";
			next;
			mes "[美容師ジョヴォヴィッチ]";
			mes "そんな特別な場面には、自分らしいヘアスタイルが欲しくなりませんか？ふさわしい染料さえあれば、髪を素敵に仕上げて差し上げますよ。";
			next;
			mes "[美容師ジョヴォヴィッチ]";
			mes "料金のことはあまり気になさらないでください。美しさを手に入れるのも、何かを得るのと同じことですから。ふふ～";
			next;
			mes "[美容師ジョヴォヴィッチ]";
			mes "＊料金とご案内＊";
			mes "・希望の髪色に合った染料アイテムを1つ";
			mes "・料金1000Zeny";
			close;

		case 3:
			mes "[美容師ジョヴォヴィッチ]";
			mes "男性でも女性でも…";
			mes "誰もが美しくある権利と義務を持っています。";
			close;
		}

	}

S_NoDye:
	if (countitem(getarg(0)) == 0) {
		mes "[美容師ジョヴォヴィッチ]";
		mes "えっ！『"+getitemname(getarg(0))+"』というアイテムが無いと、髪を"+getarg(1)+"色に染められません…";
		close;
	}
	return;
}
