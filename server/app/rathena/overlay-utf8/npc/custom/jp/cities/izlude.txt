-	script	::Sailor_izlude	-1,{
	mes "[船員]";
	mes "さあさあ皆さん！";
	mes "ご注目、ご注目！";
	mes "風を切って進む";
	mes "素敵な船に乗ってきな！";
	mes "さあ急いで急いで！";
	next;
	switch(select("ビョルン島 -> 150 Zeny.:アルベルタ港 -> 500 Zeny.:キャンセル.")) {
	case 1:
		if (Zeny < 150) {
			mes "[船員]";
			mes "150 Zeny！";
			mes "乗船料はたった150 Zenyだ！";
			close;
		}
		set Zeny, Zeny - 150;
		warp "izlu2dun",107,50;
		end;
	case 2:
		if (Zeny < 500) {
			mes "[船員]";
			mes "500 Zeny！";
			mes "乗船料はたった500 Zenyだ！";
			close;
		}
		set Zeny, Zeny - 500;
		warp "alberta",188,169;
		end;
	case 3:
		close;
	}
}
