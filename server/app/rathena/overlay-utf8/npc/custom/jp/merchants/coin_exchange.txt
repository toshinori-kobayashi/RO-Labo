//============================================================
-	script	::merchant_13_2	-1,{
	set .@n$, "["+strnpcinfo(1)+"]";
	mes .@n$;
	if (isequipped(2782) && ep13_2_rhea == 100) {
		if (!checkweight(1201,1)) {
			mes "荷物を持ちすぎているようです。";
			mes "少し倉庫に預けてから、また来てください。";
			close;
		}
		mes "こんにちは。";
		mes "何をお探しですか？";
		next;
		if (strnpcinfo(1) == "Merchant of Manuk") {
			setarray .@Items[2],12342,12343,12348;
			set .@Book,11019; //Manuk_Selling_Item
			set .@Coin,6080; //Manuk_Coin
			set .@i, select("アイテムの説明を見る:マヌクの好機:マヌクの勇気:マヌクの信仰:キャンセル");
		}
		else {
			setarray .@Items[2],12344,12345,12349;
			set .@Book,11018; //Splendide_Selling_Item
			set .@Coin,6081; //Splendide_Coin
			set .@i, select("アイテムの説明を見る:ピングイクラの果実ジャムを買う:ルキオラの蜂蜜ジャムを買う:コルヌスの涙を買う:何もしない");
		}
		mes .@n$;
		if (.@i == 1) {
			mes "こちらがアイテムの説明です。";
			close2;
			readbook .@Book,1;
			end;
		}
		else if (.@i < 5) {
			mes ""+getitemname(.@Items[.@i])+"を3個、^3131FF1コイン^000000でお売りします。";
			next;
			switch(select("購入する:購入しない")) {
			case 1:
				if (countitem(.@Coin)) {
					mes .@n$;
					mes "ご来店ありがとうございます。";
					delitem .@Coin,1;
					getitem .@Items[.@i],3;
					close;
				}
				else {
					mes .@n$;
					mes "コインが足りません。";
					close;
				}
			case 2:
				mes .@n$;
				break;
			}
		}
		mes "気が変わったら、またいらしてください。";
		close;
	}
	else {
		if (strnpcinfo(1) == "Merchant of Manuk") {
			mes "Rtt od d";
			mes "Qwo hd is d irr";
		}
		else {
			mes "BurWehAla";
			mes "tasnarAndu Ie Ru";
		}
		close;
	}
}
man_in01,286,16,1	duplicate(merchant_13_2)	Merchant of Manuk	454
spl_in01,110,326,5	duplicate(merchant_13_2)	Merchant of Splendide	439
