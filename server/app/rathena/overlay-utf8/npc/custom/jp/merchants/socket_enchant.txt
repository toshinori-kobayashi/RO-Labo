function	script	Func_Socket	{
	.@item_id = getarg(0);
	.@zeny_req = getarg(4);
	setarray .@item_req[0], getarg(5), getarg(7,0);
	setarray .@req_amount[0], getarg(6), getarg(8,0);

	disable_items;
	mes "[セイヤブレム]";
	mes "スロットを追加したいのは"+ getitemname(.@item_id) +"だな？";
	mes "よし、^FF0000"+ .@req_amount[0] +"個の"+ getitemname(.@item_req[0]) + (.@req_amount[0] > 1 && .@item_req[0] != 999 ? "":"") +"^000000、"+ (.@item_req[1] != 0 && .@req_amount[1] != 0 ? ("^FF0000"+ .@req_amount[1] +"個の"+ getitemname(.@item_req[1]) + (.@req_amount[1] > 1 && .@item_req[1] != 999 ? "":"") + "^000000、") : "") + "それに手数料の^FF0000" + (.@zeny_req >= 1000 ? (.@zeny_req /1000 +",000") : .@zeny_req) +",000 Zeny^000000を持ってきてくれ。";
	mes "ああ、それと"+ getitemname(.@item_id) +"も忘れずに持ってきてくれよ！";
	next;
	mes "[セイヤブレム]";
	if (getiteminfo(.@item_id, ITEMINFO_LOCATIONS) & EQP_HAND_R) // weapon
	{
		mes "必要なアイテムとZenyが揃っていれば、今すぐスロット追加を試せる。";
		mes "ただ、失敗する可能性もあることは知っておいてくれ。";
		mes "だから、一応警告しておくが…";
		next;
		mes "[セイヤブレム]";
		mes "武器へのスロット追加に失敗すると、^FF0000武器本体^000000と^FF0000挿入されているカードは全て消滅する^000000。";
	}
	else // armor
	{
		mes "必要な材料と手数料のZeny、そして防具本体が揃っていれば、スロット追加を進められる。";
		mes "ただしその前に、リスクについて警告しておかなければならない。";
		next;
		mes "[セイヤブレム]";
		mes "防具へのスロット追加に失敗すると、^FF0000防具本体^000000、^FF0000その強化値^000000、^FF0000挿入されているカードが全て消滅する^000000。";
	}
	mes "それでもスロット追加を試すか？";
	next;
	switch(select("スロット追加を試す:キャンセル"))
	{
		case 1:
			if ((Zeny >= .@zeny_req*1000) && countitem(.@item_req[0]) >= .@req_amount[0] && countitem(getarg(7,512)) >= .@req_amount[1] && countitem(.@item_id) > 0)// if there's no getarg(7) and getarg(8) provided, check itemcount against 0 (we still need a valid item ID though)
			{
				mes "[セイヤブレム]";
				mes "よし、それじゃあ作業開始だ！";
				mes "成功を祈っておくんだな。";
				next;
				set .@a, rand(1,100);
				if ((.@a > getarg(2)) && (.@a < getarg(3)))
				{
					specialeffect getarg(3) == 51 ? EF_LORD : EF_SANCTUARY; // only S Class items have Lord of Vermillion effect
					mes "[セイヤブレム]";
					mes "よし、成功したようだな。";
					mes "なかなかの出来だ。おめでとう！";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					getitem getarg(1),1;
					next;
					mes "[セイヤブレム]";
					mes "またな、相棒！";
					close;
				}
				else
				{
					specialeffect EF_SUI_EXPLOSION;
					mes "[セイヤブレム]";
					mes "うわっ…すまない、失敗した。";
					mes "だが俺には非は無いぞ。";
					mes "これは運、神様が決めたことだ、いいな？";
					mes "あまり落ち込まず、また次を試してくれ。";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[セイヤブレム]";
					mes "次はうまくいくよう祈ってるぞ！";
					close;
				}
			}
			else
			{
				mes "[セイヤブレム]";
				mes "スロット追加を進めたいところだが、いくつか足りないものがあるようだ。";
				mes "装備・必要な材料・Zenyがちゃんと揃っているか確認してくれ。";
				close;
			}
		case 2:
			mes "[セイヤブレム]";
			mes "少し考えたいんだな？";
			mes "よし、分かった。";
			mes "安全策ばかりじゃ人生つまらないってことだけは覚えておけよ～";
			close;
	}
}
