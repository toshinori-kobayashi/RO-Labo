//============================================================
//= Main Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//============================================================
function	script	refinemain	{
	disable_items;
	.@npc_name$ = getarg(0);
	.@features = getarg(1);
	mes "["+ .@npc_name$ +"]";
	mes "私は武具鍛冶師だ。";
	mes "あらゆる武器や防具、装備の精錬ができる。さあ、";
	mes "何を精錬したいか言ってくれ。";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<getarraysize(.@indices); ++.@i) {
		if(getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "その装備は精錬できそうにないな…";
		close;
	}
	.@part = .@indices[select(.@menu$)];

	if(!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "そこには何も";
		mes "装備していない";
		mes "ようだが。";
		emotion ET_FRET;
		close;
	}
	//Check if the item is refinable...
	if(!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "その装備は";
		mes "そもそも精錬できないようだ…";
		close;
	}
	//Check to see if the items is already +10
	if(getequiprefinerycnt(.@part) >= 10) {
		mes "["+ .@npc_name$ +"]";
		mes "もうこれ以上は";
		mes "精錬できない。これが";
		mes "限界まで極めた姿だ！";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 10;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 7;
				break;
			case 2:
				.@safe = 6;
				break;
			case 3:
				.@safe = 5;
				break;
			case 4:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 40;
					break;
				case 2:
					.@price = .@price * 50;
					break;
				case 3:
					.@price = .@price * 2;
					break;
				case 4:
					.@price = .@price * 2;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else{
		.@safe = 4;
	}

	// TODO: What was this? Maybe shadow?
	// case 5: .@price = .@price * 10; break;

	if(.@features != 1) {
		mes "["+ .@npc_name$ +"]";
		mes "精錬には";
		mes "^003366"+getitemname(.@material)+"^000000 が1つと";
		mes "手数料 " + .@price + " Zeny が必要だ。";
		mes "本当に続けるか？";
		next;
		if(select("はい:いいえ") == 2){
			mes "["+ .@npc_name$ +"]";
			mes "そうか…";
			mes "急ぐことはない。";
			mes "ゆっくり考えるといい。";
			close;
		}
		if(getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "しまった！このまま精錬を続けると";
			mes "壊れてしまうかもしれない。";
			switch(.@material) {
			case 985:
				mes "そうなれば^FF0000この防具^000000も^FF0000付いているカードや特殊効果^000000も^FF0000すべて失われてしまう^000000。";
				break;
			default:
				mes "そうなれば^FF0000武器そのもの^000000も^FF0000挿さっているカード^000000も、";
				mes "追加された特殊効果もすべて失ってしまうぞ。";
				break;
			}
			next;
			mes "["+getarg(0)+"]";
			mes "これ以上、はっきりとは言えん。";
			mes "一度壊れてしまった武器は、";
			mes "二度と元には戻らない。";
			mes "本当に";
			mes "^FF0000この武器を永遠に失う^000000かもしれないんだぞ。";
			mes "それでも精錬を続けるか？";
			next;
			if(select("はい:いいえ") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "だな…";
				mes "腕利きの精錬師とはいえ、俺だって失敗することはあるさ。";
				close;
			}
		}
		if((countitem(.@material) < 1) || (Zeny < .@price)) {
			mes "["+ .@npc_name$ +"]";
			mes "Zenyが足りないようだ。";
			mes ""+getitemname(.@material)+"が足りないようだ。";
			mes "また取ってくるといい。";
			mes "俺は一日中ここにいるからな。";
			close;
		}
		Zeny = Zeny-.@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "ちょっと待て…";
			mes "俺を馬鹿にしてるのか？！";
			mes "見てない隙に品物をすり替えたな！とっとと出て行け！";
			close;
		}

		if(getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			mes "["+ .@npc_name$ +"]";
			emotion (!rand(5))?ET_MONEY:ET_HUK;
			.@lose = rand(1,3);
			if (.@lose == 1) {
				mes "なんてことだ！";
				mes "くそっ！また失敗か！";
				mes "本当にすまない。だが、習うより慣れろと言うだろう？";
				mes "……な？ははは…";
			} else if(.@lose == 2) {
				mes "うわあああ！";
				mes "壊れた！";
				mes "す、すまない！";
			} else {
				mes "しまった！";
				mes "これ以上の鍛錬には";
				mes "耐えられなかったか！";
				mes "すまないな…";
			}
			close;
		}
		mes "["+getarg(0)+"]";
		successrefitem .@part;
		emotion ET_SMILE;
		.@win = rand(1,3);
		if (.@win == 1) {
			mes "完璧だ！";
			mes "ヘヘッ！";
			mes "また";
			mes "この俺様の";
			mes "完璧な仕事よ～";
		} else if(.@win == 2) {
			mes "成功だ…！";
			mes "今日もまた、俺の";
			mes "とびきりの腕前が";
			mes "冴え渡ったな。";
		} else {
			mes "ヘヘッ！";
			mes "終わったぞ。";
			mes "きっと満足の";
			mes "いく出来だろう。";
			mes "まさに完璧よ～";
		}
		close;
	}

// New Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "安全圏までなら確実に精錬できる。好きな回数を選んでくれ。";
		next;
		.@menu2 = select("安全圏まで頼む。","回数は自分で決める。","やっぱりやめておく…");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "何回精錬すればいい？";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 10) {
			mes "["+ .@npc_name$ +"]";
			mes "その回数は精錬できない。";
			close;
		}
		if(.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "安全圏を" + .@refinecheck + "回超えて精錬することになる。装備が壊れるかもしれないが、それでもいいか？";
			next;
			if(select("かまわない…","やめておく…") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "そう言うなら…仕方ないな。";
				close;
			}
		}
		break;
	case 3:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "そう言うなら…仕方ないな。";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "" + .@refinecnt + "回分の" + getitemname(.@material) + "と" + .@fullprice + " Zeny が必要だ。よろしいか？";
	next;
	if(select("はい","いいえ…") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "そう言うなら…仕方ないな。";
		close;
	}
	if(countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "それっぽっちか？残念だが、これ以上は安くできない。俺の立場にもなってくれ。";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "おい…何も装備していないじゃないか…";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
				callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt) || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "カンッ…おいおい、俺がそんなに馬鹿に見えるか？！";
			mes "すり替えたな…";
			mes "ハンマーでぶちのめされる前に出て行け！！";
			close;
		} 
		mes "カンッ、カンッ！！！";
		if(.@menu2 == 2 && getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			emotion ET_HUK;
			mes "["+ .@npc_name$ +"]";
			mes "うわあああ！！すまない…こうなるかもしれないと言っただろう…";
			.@refinecnt = .@refinecnt - 1;
			if(.@refinecnt == 0) close;
			mes "使わなかった分のZenyと材料を返す…";
			getitem .@material,.@refinecnt;
			.@fullprice = .@refinecnt * .@price;
			Zeny = Zeny + .@fullprice;
			close;
		}
		successrefitem .@part;
		emotion ET_BEST;
		.@refinecnt = .@refinecnt - 1;
		.@refinerycnt = getequiprefinerycnt(.@part);
		next;
	}
	mes "["+ .@npc_name$ +"]";
	mes "終わったぞ…また来てくれ。";
	close;
}

function	script	phramain	{
	if (checkweight(1201,1) == 0) {
		mes "- ちょっと待った！！ -";
		mes "- 今、あなたは -";
		mes "- 荷物を持ちすぎている -";
		mes "- 少し減らしてから -";
		mes "- もう一度試してほしい -";
		close;
	}
	.@npc_name$ = getarg(0);
	mes "["+ .@npc_name$ +"]";
	mes "武器を鍛えるための金属を";
	mes "2種類扱っている。";
	mes "レベル1の武器には^007777フラコン^000000、";
	mes "レベル2の武器には^007777エンベルタコン^000000";
	mes "が要る。";
	next;
	switch(select("フラコン - 200 Zeny:エンベルタコン - 1000 Zeny:他の金属について尋ねる")) {
	case 1:
		.@material = 1010;
		.@price = 200;
		break;
	case 2:
		.@material = 1011;
		.@price = 1000;
		break;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "他の金属？";
		mes "上位の武器や防具を強化するには特別な金属が要る。だが、オリデオコンやエルニウムというのは";
		mes "そう簡単には手に入らないものでな…";
		close;
	}
	mes "["+ .@npc_name$ +"]";
	mes "さて、いくつ買う？";
	mes "いらないなら「0」と入力してくれ。";
	next;
	while(1) {
		input .@input;
		if (.@input == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "取引は";
			mes "中止だ。";
			close;
		}
		else if (.@input < 0 || .@input > 500) {
			mes "["+ .@npc_name$ +"]";
			mes "いいか、買えるのは";
			mes "500個までだ。";
			mes "それ以上は無理だぞ、";
			mes "わかったか？よし。";
			next;
		}
		else {
			break;
		}
	}
	.@sell = .@input * .@price;
	if (Zeny < .@sell) {
		mes "["+ .@npc_name$ +"]";
		mes "うーん…";
		mes "それだけのZenyじゃ";
		mes "足りないぞ、";
		mes ""+ .@input +"個分にはな。";
		close;
	}
	if (checkweight(.@material,.@input) == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "うーん…";
		mes "所持品に空きがなければ渡せないな。余分な物はカプラの倉庫に預けて、また来てくれ。";
		close;
	}
	getitem .@material,.@input;
	Zeny = Zeny-.@sell;
	mes "["+ .@npc_name$ +"]";
	mes "はい、どうぞ！";
	mes "ご利用";
	mes "ありがとう。";
	close;
}

function	script	orimain	{
	if (checkweight(1201,1) == 0) {
		mes "- ちょっと待った！！ -";
		mes "- 今、あなたは -";
		mes "- 荷物を持ちすぎている -";
		mes "- 少し減らしてから -";
		mes "- もう一度試してほしい -";
		close;
	}
	.@npc_name$ = getarg(0);
	mes "["+ .@npc_name$ +"]";
	mes "原石オリデオコンや";
	mes "原石エルニウムを";
	mes "精製できるぞ。原石を";
	mes "5個集めれば";
	mes "純粋な物が1個できる。";
	next;
	switch(select("オリデオコンを作る:エルニウムを作る:精霊石について尋ねる")) {
	case 1:
		if (countitem(756) > 4) {
			delitem 756,5;  //Oridecon_Stone
			getitem 984,1; // Oridecon
			mes "["+ .@npc_name$ +"]";
			mes "オリデオコンだ。";
			mes "また来たい時に";
			mes "いつでも来るといい。";
			close;
		}
		else {
			mes "["+ .@npc_name$ +"]";
			mes "冗談だろう？";
			mes "純粋なオリデオコンを作るには原石オリデオコンが5個要ると、今言ったばかりだぞ。";
			close;
		}
	case 2:
		if (countitem(757) > 4) {
			delitem 757,5;  //Elunium_Stone
			getitem 985,1; // Elunium
			mes "["+ .@npc_name$ +"]";
			mes "エルニウムだ。";
			mes "また来たい時に";
			mes "いつでも来るといい。";
			close;
		}
		else {
			mes "["+ .@npc_name$ +"]";
			mes "冗談だろう？";
			mes "純粋なエルニウムを作るには原石エルニウムが5個要ると、今言ったばかりだぞ。";
			close;
		}
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "精霊石…？";
		mes "石職人を20年もやっていれば、色々な話を耳にするもんでな。聞くところによると";
		mes "4種類あるらしい。";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "精霊石にはそれぞれ、地・風・水・火のいずれかの属性が宿っているそうだ。";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "鍛冶の際に精霊石を武器と組み合わせれば、その武器は石と同じ属性を帯びるという。";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "言うまでもないが、そうした属性武器を作るにはそれなりの鍛冶の腕が要る。";
		close;
	}
}

// Equipment Repair Function
//============================================================
function	script	repairmain	{
	.@repairprice = 5000;
	.@npc_name$ = getarg(0);
	mes "["+ .@npc_name$ +"]";
	mes "よお！";
	mes "何か直したい物が";
	mes "あるんじゃないか？";
	mes "修理の腕なら";
	mes "任せておけ！";
	next;
	switch(select("実は、直したい物があるんだ…:今のところは何もない。")) {
	case 1:
		.@checkitem = 1;
		while (1) {
			if (getbrokenid(.@checkitem) == 0) {
				break;
			}
			.@checkitem = .@checkitem+1;
		}
		.@checkitem = .@checkitem-1;
		if (!.@checkitem) {
			mes "["+ .@npc_name$ +"]";
			mes "おお、こいつは驚いた！";
			mes "よほど物を大事にしているんだな。壊れている物はひとつもないぞ！";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "みんながお前さんみたいだったら、俺は仕事にあぶれちまうな！！ハハッ～！";
			close;
		}
		mes "["+ .@npc_name$ +"]";
		mes "うーん…";
		mes "そうだな…";
		mes "お前さんの持ち物のうち、";
		mes "" + .@checkitem + "個が壊れているな。";
		mes "修理するか？";
		next;
		.@totalcost = .@repairprice*.@checkitem;
		mes "["+ .@npc_name$ +"]";
		mes "修理1件につき" + .@repairprice + " Zeny。壊れた物を全部直すと" + .@totalcost + " Zeny になるが、修理するか？";
		next;
		switch(select("はい:いいえ")) {
		case 1:
			if (Zeny < .@totalcost) {
				mes "["+ .@npc_name$ +"]";
				mes "おいおい…";
				mes "請求書を受け取る前に財布の中身を確かめな！Zenyが足りなきゃ何も直せないぞ。";
				close;
			}
			.@checkitem2 = 1;
			while (1) {
				if (getbrokenid(.@checkitem2) == 0) {
					break;
				}
				.@checkitem2 = .@checkitem2+1;
			}
			.@checkitem2 = .@checkitem2-1;
			if (.@checkitem == .@checkitem2) {
				Zeny = Zeny-.@totalcost;
				while (.@checkitem) {
					repair(.@checkitem);
					.@checkitem = .@checkitem-1;
				}
				mes "["+ .@npc_name$ +"]";
				mes "よし！終わったぞ。もう少し大事に扱ってやれよ。道具にも命があるんだからな。";
				close;
			}
			else {
				mes "["+ .@npc_name$ +"]";
				mes "ん？何かおかしいな。直したい物は装備してから出直してきてくれ。";
				close;
			}
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "まあ、俺は別に構わないが、壊れたまま放っておくのは良くないぞ。早めに直しておいた方がいい！";
			close;
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "ホッホッホ…";
		mes "直したい物が";
		mes "ないんじゃ";
		mes "俺のところに";
		mes "用はないだろう。";
		close;
	}
}
