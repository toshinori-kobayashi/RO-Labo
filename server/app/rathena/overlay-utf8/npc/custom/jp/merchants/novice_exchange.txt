function	script	F_PotExchange	{
	.@item_req = getarg(0);
	.@req_amount = getarg(1);

	if (countitem(.@item_req) < .@req_amount) {
		mes "[商人]";
		mes "おや、約束してくれた";
		mes ""+ getitemname(.@item_req) +"は";
		mes "どこにあるんだい？";
		mes ""+ getitemname(.@item_req) +"を持ってきておくれ！";
		close;
	}
	else {
		mes "[商人]";
		mes "よし、確認させて";
		mes ""+ getitemname(.@item_req) +"をいくつ";
		mes "持っているか…ふむ…";
		next;
		mes "[商人]";
		mes "君は";
		mes "合計"+ countitem(.@item_req) +"個の"+ getitemname(.@item_req) +"を持っているね。";
		mes "それなら合計で";
		mes ""+ (countitem(.@item_req)/.@req_amount) +"個の赤ポーションと交換できるよ。";
		next;
		mes "[商人]";
		mes "どうだい？";
		mes "取引成立かな？";
		next;
		if (select("取引成立:取引しない") == 1) {
			mes "[商人]";
			mes "欲しい赤ポーションの";
			mes "正確な数は";
			mes "もう決まっているんだろう？";
			mes ""+ getitemname(.@item_req) +"と交換するんだったね？";
			next;
			mes "[商人]";
			mes "そうだな、一度に";
			mes "最低1個から";
			mes "最大100個まで";
			mes "赤ポーションと交換できるよ。もし";
			mes "やめたくなったら「0」と入力してくれ。";
			next;
			while(1) {
				input .@input,0,101;
				if (.@input == 0) {
					mes "[商人]";
					mes "何だって…？";
					mes "それならどうして";
					mes "話しかけてきたんだい？";
					mes "ずいぶん優柔不断なんだねえ。";
					close;
				}
				else if (.@input > 100) {
					mes "[商人]";
					mes "一度に100個より";
					mes "多くは渡せないよ。";
					mes "もう一度";
					mes "やり直しておくれ。";
					next;
				}
				else {
					break;
				}
			}
			.@put_out_item = .@input * .@req_amount;
			if (countitem(.@item_req) < .@put_out_item) {
				mes "[商人]";
				mes "おっと、入力した数がおかしいようだね。交換できる赤ポーションの数をもう一度確認してみておくれ。";
				close;
			}
			mes "[商人]";
			mes "はい、どうぞ！";
			mes "渡した赤ポーションの数を";
			mes "確認してみておくれ。ありがとう、いい取引だったよ～";
			delitem .@item_req,.@put_out_item; //Shell
			getitem 501,.@input; //Red_Potion
			close;
		}
		mes "[商人]";
		mes "ふうん～";
		mes "分かったよ。";
		mes "でも君みたいな";
		mes "冒険者には";
		mes "赤ポーションの方が役に立つと思うけどねえ？";
		close;
	}
}
