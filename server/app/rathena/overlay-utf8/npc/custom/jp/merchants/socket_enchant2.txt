					mes "また次の機会に挑戦してください。";

					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[レアブレム]";
					mes "また会いましょうね！";
					close;
				}
			}
			else
			{
				mes "[レアブレム]";
				mes "何を考えているんですか？必要なものが全部揃っていませんよ！";
				mes "早く持ってきてください！";
				close;
			}
		case 2:
			mes "[レアブレム]";
			mes "また今度お待ちしています。";
			close;
	}
