		mes "[ホテル従業員]";

//======================= Inn Function ==============================
function	script	F_InnMaid	{
	.@npc_name$ = getarg(0);

	mes .@npc_name$;
	mes "ようこそ、";
	mes getarg(1) + ".";
	mes "ご用件は何でしょうか？";
	next;
	switch(select("セーブする:休憩する -> 5000 Zeny:キャンセル")) {
	case 1:
		mes .@npc_name$;
		mes "リスポーン地点を";
		mes "設定いたしました。";
		mes "ありがとうございます、";
		mes "またお越しください。";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close;
	case 2:
		mes .@npc_name$;
		if(Zeny < 5000){
			mes "申し訳ございませんが、利用料は5,000 Zenyです。次はチェックインに十分なお金をご用意くださいね。";
			close;
		}
		mes "ありがとうございます。";
		mes "ごゆっくり";
		mes "おくつろぎください～";
		close2;
		set Zeny,Zeny - 5000;
		percentheal 100,100;
		return;
	case 3:
		close;
	}
}
