prontera,73,140,0	script	牛乳商人::Milk Vendor	86,{
	if (checkweight(1201,1) == 0) {
		mes "^3355FFちょっと待ってください！";
		mes "荷物を持ちすぎているので";
		mes "今はサービスを";
		mes "提供できません。";
		mes "余分な荷物はカプラ倉庫に";
		mes "預けてから、また来てください～";
		close;
	}
	mes "[牛乳商人]";
	mes "やあやあ…";
	mes "空き瓶1個と";
	mes "15Zenyを";
	mes "持ってきてくれたら、";
	mes "牛乳1個と交換するよ。";
	mes "どうだい？";
	next;
	if (select("空き瓶を全部交換する。:キャンセル") == 1) {
		if (countitem(713) <= 0) {
			mes "[牛乳商人]";
			mes "おいおい…";
			mes "空き瓶を";
			mes "一つも持ってないじゃないか。";
			mes "それじゃあ牛乳を";
			mes "渡すわけには";
			mes "いかないんだよ…";
			close;
		}
		set .@bottles,countitem(713);
		set .@total_weight,.@bottles * 50;
		set .@total_cost,.@bottles * 15;
		if (Zeny < .@Total_cost) {
			mes "[牛乳商人]";
			mes "おっと～！";
			mes "空き瓶を全部";
			mes "牛乳に交換するには";
			mes "Zenyが";
			mes "足りません。";
			mes "最低" + .@total_cost + "Zenyは必要です。";
			close;
		}
		if (MaxWeight - Weight < .@total_weight) {
			mes "[牛乳商人]";
			mes "うーん…";
			mes "この牛乳を全部";
			mes "渡す前に、";
			mes "所持品に";
			mes "少し空きを";
			mes "作ってもらえないかい？";
			close;
		}
		set Zeny, Zeny-.@total_cost;
		delitem 713,.@bottles; //Empty Bottles
		getitem 519,.@bottles; //Milk
		close;
	}
	close;
}
