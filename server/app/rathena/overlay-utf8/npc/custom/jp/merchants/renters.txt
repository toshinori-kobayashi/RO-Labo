// PecoPeco Breeder (for Knights) :: breeder
//============================================================
prontera,55,350,5	script	ペコペコ飼育員#knt::Peco Peco Breeder#knt	105,{
	if(Upper==0) set .@price,2500; //Normal Peco - default price
	if(Upper==1) set .@price,2500; //Armored Peco
	if(Upper==2) set .@price,2500; //Baby Peco

	mes "[ペコペコ飼育員]";
	if (BaseJob == Job_Knight && Class < Job_Rune_Knight) {
		mes "ようこそ。";
		mes "誇り高きナイトよ、";
		mes "ペコペコを";
		mes "借りたいか？レンタル";
		mes "料金は"+.@price+" Zenyだ。";
		next;
		switch(select("ペコペコを借りる:キャンセル")) {
		case 1:
			if (Zeny < .@price) {
				mes "[ペコペコ飼育員]";
				mes "Zenyが";
				mes "足りないようだな。";
				mes "まさか…";
				mes "一文無しか？";
				close;
			}
			else if(getskilllv("KN_RIDING") == 0) {
				mes "[ペコペコ飼育員]";
				mes "悪いが、お前さんは";
				mes "まだこのサービスを";
				mes "利用できない。まずは";
				mes "ペコペコ騎乗を覚えてくるといい。";
				close;
			}
			else if(checkriding()) {
				mes "[ペコペコ飼育員]";
				mes "もう既に";
				mes "ペコペコに";
				mes "乗っているじゃないか。";
				close;
			}
			else if(ismounting()) {
				mes "[ペコペコ飼育員]";
				mes "キャッシュマウントを解除してくれ。";
				close;
			}
			set Zeny, Zeny-.@price;
			setriding;
			close;
		case 2:
			mes "[ペコペコ飼育員]";
			mes "そうか。";
			mes "それじゃ、";
			mes "良い一日を。";
			close;
		}
	} else {
		mes "悪いが、この";
		mes "ペコペコは";
		mes "ナイトと";
		mes "ロードナイト専用だ。";
		close;
	}
}

// Grand PecoPeco Breeder (for Crusaders)
//============================================================
prontera,232,318,3	script	ペコペコ飼育員#cru::Peco Peco Breeder#cru	105,{
	if(Upper==0) set .@price,3500; //Normal Peco - default price
	if(Upper==1) set .@price,3500; //Armored Peco
	if(Upper==2) set .@price,3500; //Baby Peco

	mes "[ペコペコ飼育員]";
	if (BaseJob == Job_Crusader && Class < Job_Rune_Knight) {
		if(Upper != 1 ) mes "ようこそ、クルセイダー殿。";
		else mes "ようこそ、パラディン殿。";
		mes "あなたのために";
		mes "特別なペコペコを";
		mes "用意した。借りるには";
		mes "レンタル料は"+.@price+" Zenyだ。";
		next;
		switch(select("ペコペコを借りる:やめる")) {
		case 1:
			if (Zeny < .@price) {
				mes "[ペコペコ飼育員]";
				mes "Zenyが";
				mes "足りないようだ。";
				mes "ペコペコが";
				mes "欲しいなら";
				mes ""+.@price+" Zeny持ってきてくれ…";
				close;
			}
			else if(getskilllv("KN_RIDING") == 0) {
				mes "[ペコペコ飼育員]";
				mes "貸す前に、まず";
				mes "ペコペコに乗る術を";
				mes "覚えてもらわないとな。";
				close;
			}
			else if(checkriding()) {
				mes "[ペコペコ飼育員]";
				mes "もう既に";
				mes "ペコペコに乗っているぞ。";
				close;
			}
			else if(ismounting()) {
				mes "[ペコペコ飼育員]";
				mes "キャッシュマウントを解除してくれ。";
				close;
			}
			set Zeny, Zeny-.@price;
			setriding;
			close;
		case 2:
			mes "[ペコペコ飼育員]";
			mes "またな。";
			close;
		}
	} else {
		mes "何か用か？";
		mes "このペコペコレンタルは";
		mes "クルセイダーと";
		mes "パラディン専用だと";
		mes "承知しておいてくれ。";
		close;
	}
}

// Falcon Master :: breeder
//============================================================
hu_in01,381,304,5	script	ファルコン飼育員#hnt::Falcon Breeder#hnt	105,{
	if(Upper==0) set .@price,2500; //Normal Falcon - default price
	if(Upper==1) set .@price,2500; //Scarf Falcon
	if(Upper==2) set .@price,2500; //Baby Falcon

	mes "[ファルコン飼育員]";
	if (BaseJob == Job_Hunter) {
		if (checkwug()) {
			mes "うーん…";
			mes "ワーグを解除しないと";
			mes "ファルコンは貸せないぞ！";
			close;
		}
		mes "ファルコンが要るか？";
		mes "頼れる猛禽を";
		mes "たった";
		mes ""+.@price+" Zenyで貸してやるぞ～";
		next;
		switch(select("ファルコンを借りる:キャンセル")) {
		case 1:
			if (checkwug()) {
				mes "[ファルコン飼育員]";
				mes "うーん…";
				mes "ワーグを解除しないと";
				mes "ファルコンは貸せないぞ！";
				close;
			}
			if (Zeny < .@price) {
				mes "[ファルコン飼育員]";
				mes "なんだこれは？";
				mes "Zenyが";
				mes "足りないだと？！";
				mes "モンスターより";
				mes "先にZenyでも";
				mes "狩ってきたらどうだ～";
				close;
			}
			else if(getskilllv("HT_FALCON") == 0) {
				mes "[ファルコン飼育員]";
				mes "やれやれ～";
				mes "まずはファルコンの";
				mes "扱い方を";
				mes "覚えてこい！扱えない";
				mes "やつには";
				mes "貸せないからな。";
				close;
			}
			else if(checkfalcon()) {
				mes "[ファルコン飼育員]";
				mes "うーん…";
				mes "もうファルコンを";
				mes "持っているじゃないか。";
				mes "ほら、そこに見えるだろう。";
				close;
			}
			set Zeny, Zeny-.@price;
			setfalcon;
			close;
		case 2:
			mes "[ファルコン飼育員]";
			mes "ま、待てよ、";
			mes "どこ行くんだ？この";
			mes "ファルコンは極上だぜ、";
			mes "保証するって！なあ、頼むよ～";
			close;
		}
	} else {
		mes "青二才が！";
		mes "ファルコンは";
		mes "ハンターと";
		mes "スナイパーだけの物だ、わかったか？";
		mes "…ヘヘッ、うらやましいか？";
		close;
	}
}
