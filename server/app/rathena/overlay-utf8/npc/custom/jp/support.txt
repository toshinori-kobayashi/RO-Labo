//============================================================
//= 日本語サポート NPC（このサーバ独自）
//============================================================
//= 用途:
//=   プロンテラに常駐し、回復・主要都市への移動・サーバ説明を
//=   日本語で提供する検証用の窓口 NPC。上流スクリプトの
//=   差し替えではなく、このサーバだけの追加 NPC。
//=   読み込みは overlay-utf8/npc/scripts_custom.conf の
//=   "npc: npc/custom/jp/support.txt" で行う。
//=
//= 文字コード:
//=   このファイルは UTF-8 で管理し、Docker イメージのビルド時に
//=   iconv で CP932 へ変換されてから /opt/rathena/ に配置される。
//=   したがって編集は必ず UTF-8 / 改行 LF / BOM なしで行うこと。
//=   CP932 に存在しない文字（U+301C 波ダッシュ、絵文字など）を
//=   書くとビルドが失敗する。波ダッシュは「～」(U+FF5E)、
//=   マイナスは「－」(U+FF0D) を使うこと。
//=
//= 注意:
//=   select() の選択肢は半角コロンで区切られるため、
//=   選択肢の文言に半角コロンを含めないこと。
//=
//= 座標: prontera,160,180（噴水の南東。既存 NPC と重ならない位置）
//============================================================

prontera,160,180,4	script	サポート職員	4_F_KAFRA1,{
	mes "[サポート職員]";
	mes "いらっしゃいませ。";
	mes "こちらは検証用サーバの案内窓口です。";
	mes "ご用件をお選びください。";
	next;
	switch (select("体力と精神力を全回復する:主要な街へ移動する:このサーバについて:やめる")) {

	case 1:
		percentheal 100,100;
		mes "[サポート職員]";
		mes "体力と精神力を全回復しました。";
		mes "どうぞお気をつけて。";
		close;

	case 2:
		mes "[サポート職員]";
		mes "移動先の街をお選びください。";
		mes "移動は無料です。";
		next;
		switch (select("プロンテラ:ゲフェン:フェイヨン:モロク:アルベルタ:イズルード:アルデバラン:コモド:やめる")) {
		case 1: warp "prontera",155,183; end;
		case 2: warp "geffen",119,59; end;
		case 3: warp "payon",179,100; end;
		case 4: warp "morocc",156,93; end;
		case 5: warp "alberta",28,234; end;
		case 6: warp "izlude",128,114; end;
		case 7: warp "aldebaran",140,131; end;
		case 8: warp "comodo",209,143; end;
		case 9:
			mes "[サポート職員]";
			mes "かしこまりました。";
			mes "またのご利用をお待ちしております。";
			close;
		}

	case 3:
		mes "[サポート職員]";
		mes "このサーバについてご案内します。";
		mes "動作モードは Pre-Renewal です。";
		mes "三次職や新しい街は実装されていません。";
		next;
		mes "[サポート職員]";
		mes "経験値の取得量は通常の１０倍、";
		mes "アイテムの入手しやすさは通常の５倍です。";
		mes "カードの入手しやすさは種類によって";
		mes "１０倍～１００倍に調整されています。";
		next;
		mes "[サポート職員]";
		mes "不具合のご報告やお問い合わせ先は、";
		mes "配布元の README に記載しています。";
		mes "そちらをご確認ください。";
		close;

	case 4:
		mes "[サポート職員]";
		mes "かしこまりました。";
		mes "またのご利用をお待ちしております。";
		close;
	}
}
